import { getRequestClientIp, normalizeClientIp, type ClientIpTrustConfig } from "./request-client-ip";

export type EntryIpTrustConfig = ClientIpTrustConfig | {
  trustMode: "vercel-native";
  deploymentSource: "vercel";
};

/** Entry trust is explicit; a storage/WAF provider never selects a proxy. */
export function entryIpTrustConfig(env: NodeJS.ProcessEnv = process.env): EntryIpTrustConfig {
  const trustMode = env.ENTRY_CLIENT_IP_TRUST_MODE;
  const deploymentSource = env.ENTRY_CLIENT_IP_DEPLOYMENT_SOURCE;
  if (trustMode === "vercel-native") {
    // Explicit Preview-only rollout. Runtime metadata is server-side, never headers.
    const expectedProject = env.ENTRY_CLIENT_IP_EXPECTED_PROJECT_ID;
    const verified = deploymentSource === "vercel"
      && env.ENTRY_CLIENT_IP_TOPOLOGY === "direct-vercel"
      && env.VERCEL === "1" && env.VERCEL_ENV === "preview"
      && Boolean(expectedProject) && expectedProject === env.VERCEL_PROJECT_ID
      && env.RATE_LIMIT_PROVIDER === "upstash_redis";
    return verified ? { trustMode, deploymentSource: "vercel" }
      : { trustMode: "none", deploymentSource: "none" };
  }
  if (trustMode === "runtime" && deploymentSource === "node") {
    return { trustMode, deploymentSource };
  }
  if ((trustMode === "cloudflare" && deploymentSource === "cloudflare")
    || (trustMode === "trusted-proxy" && deploymentSource === "vercel")) {
    return { trustMode, deploymentSource, ingressSecret: env.ENTRY_CLIENT_IP_INGRESS_SECRET };
  }
  return { trustMode: "none", deploymentSource: "none" };
}

/** Native IPv6 sources share a /64 bucket; mapped IPv4 shares its IPv4 bucket. */
export function nativeBucketIdentity(value: string | null) {
  const ip = normalizeClientIp(value);
  if (!ip || !ip.includes(":")) return ip;
  const [head, tail] = ip.split("::");
  const left = head ? head.split(":") : [];
  const right = tail ? tail.split(":") : [];
  const groups = tail === undefined ? left
    : [...left, ...Array<string>(8 - left.length - right.length).fill("0"), ...right];
  const words = groups.map(group => Number.parseInt(group, 16));
  if (words.slice(0, 5).every(word => word === 0) && words[5] === 0xffff) {
    return [words[6]! >> 8, words[6]! & 255, words[7]! >> 8, words[7]! & 255].join(".");
  }
  return `${words.slice(0, 4).map(word => word.toString(16)).join(":")}::`;
}

/** 回傳真實正規化 IP；業務授權與封鎖名單不得拿 IPv6 /64 bucket 代替地址。 */
export function getEntryClientIp(request: Request, config: EntryIpTrustConfig = entryIpTrustConfig()) {
  if (config.trustMode === "vercel-native") {
    // 每次重新核對部署 metadata；傳入 config 不能繞過 Preview 限制。
    return entryIpTrustConfig().trustMode === "vercel-native"
      ? normalizeClientIp(request.headers.get("x-vercel-forwarded-for")) : null;
  }
  return getRequestClientIp(request, config);
}

import { entryIpTrustConfig } from "./entry-client-ip";
import { hasStrongIngressSecret } from "./request-client-ip";

/** 僅回報固定設定狀態；不輸出 secret、URL、專案 ID，也不宣稱網路拓撲已通過探測。 */
export function entryReadiness(env: NodeJS.ProcessEnv = process.env) {
  const config = entryIpTrustConfig(env);
  if (env.VERCEL_ENV === "production") return { ready: false, reason: "production_topology_unverified" } as const;
  if (config.trustMode === "none") return { ready: false, reason: "entry_configuration_missing_or_invalid" } as const;
  if (env.RATE_LIMIT_PROVIDER === "upstash_redis" && !(env.UPSTASH_REDIS_REST_URL && env.UPSTASH_REDIS_REST_TOKEN)) {
    return { ready: false, reason: "shared_rate_limit_not_configured" } as const;
  }
  if (config.trustMode === "vercel-native") return { ready: true, reason: "preview_native_configured" } as const;
  const local = env.NODE_ENV !== "production" && !env.VERCEL;
  if (!local && env.VERCEL_ENV !== "preview") return { ready: false, reason: "deployment_topology_unverified" } as const;
  if (config.trustMode === "runtime") return local
    ? { ready: true, reason: "local_runtime_configured" } as const
    : { ready: false, reason: "runtime_address_unavailable_on_deployment" } as const;
  if (!hasStrongIngressSecret(config.ingressSecret)) return { ready: false, reason: "ingress_proof_not_configured" } as const;
  // 遠端 app 限流必須使用共享儲存；沒有 WAF 規則證據時不得宣稱 WAF 已覆蓋路由。
  if (env.RATE_LIMIT_PROVIDER !== "upstash_redis" && !(local && env.RATE_LIMIT_PROVIDER === "memory")) {
    return { ready: false, reason: "shared_rate_limit_provider_required" } as const;
  }
  return { ready: true, reason: "proof_ingress_configured" } as const;
}

import { NextResponse } from "next/server";
import { checkRateLimit as checkLegacyRateLimit } from "./rate-limit";
import { entryIpTrustConfig, getEntryClientIp, nativeBucketIdentity, type EntryIpTrustConfig } from "./entry-client-ip";
export { entryIpTrustConfig } from "./entry-client-ip";
import { rateLimitRequestWithIdentity } from "./live-chat-request-security";

/** Only WP11 callers opt in. Keep the original body untouched for redemption. */
export async function checkRateLimit(
  request: Request, key: string, limit: number, windowMs: number,
  config: EntryIpTrustConfig = entryIpTrustConfig(),
) {
  const native = config.trustMode === "vercel-native";
  // No fallback: direct Vercel traffic cannot rely on Cloudflare WAF or per-instance memory.
  const resolvedIp = getEntryClientIp(request, config);
  const ip = native ? nativeBucketIdentity(resolvedIp) : resolvedIp;
  if (!ip) {
    return NextResponse.json({ error: "Request source unavailable" }, {
      status: 503,
      headers: { "Retry-After": "30", "Cache-Control": "no-store" },
    });
  }
  return checkLegacyRateLimit(rateLimitRequestWithIdentity(request, ip), native ? `entry-vercel-native:${key}` : key, limit, windowMs);
}

# WP-11 非 Production 入口部署檢查

本文件僅涵蓋本機與獨立 Preview。Production 拓撲尚未驗證，屬 release blocker，不可將 Preview 證據視為 Production 部署核可。

| 環境 | 來源設定 | 限流儲存 | 驗證要求 |
| --- | --- | --- | --- |
| Vercel 獨立 Preview／staging Preview | ENTRY_CLIENT_IP_TRUST_MODE=vercel-native、ENTRY_CLIENT_IP_DEPLOYMENT_SOURCE=vercel、ENTRY_CLIENT_IP_TOPOLOGY=direct-vercel；EXPECTED_PROJECT_ID 必須匹配平台 VERCEL_PROJECT_ID，VERCEL_ENV 必須 preview | upstash_redis；連線資訊由核准 process／平台 secret provider 注入 | `/api/health/entry` 200 且兩個實際出口來源 probe 通過；不能用自行偽造的 header 取代部署 probe |
| 本機 owned ingress | cloudflare/cloudflare proof 模式；代理覆寫來源 IP 及 proof header | memory 僅本機單程序 | localhost ingress 使用合成 proof，確認直接 origin／偽造 proof 被拒，瀏覽器流程成功 |
| 本機提供 request.ip 的 runtime adapter | runtime/node | 本機 memory | 必須證明 adapter 實際提供 ip；Next Request 不會因 header 存在而自動獲得 runtime IP |
| Preview 上受控 proof ingress | cloudflare/cloudflare 或 trusted-proxy/vercel；ENTRY_CLIENT_IP_INGRESS_SECRET 經平台注入 | upstash_redis | 實測代理覆寫 proof 與來源 IP；不能假設 fronting proxy 的 egress 就是使用者 IP |

完整變數名稱為 `ENTRY_CLIENT_IP_EXPECTED_PROJECT_ID`。不得讀取 `.env*`、列舉 secret store 或輸出憑證。Redis 設定只由應用程式在執行時使用；readiness 僅回報固定類別，不回傳值。

`/api/health/entry` 的 `topologyProbeRequired: true` 表示此檢查只證明設定格式與必要欄位存在，未證明外部網路、Redis 可連線或來源覆寫行為。實際 smoke 另須包含 portal、voucher、admission、referral、chat GET/POST、private chat 與 purchase broadcasts；只登入頁 200 不足以驗收。

已明確設定 ENTRY 模式的部署，入口及直播聊天使用同一個 ENTRY proof 與解析器。無效的 ENTRY 設定不退回舊聊天設定。未設定 ENTRY 時僅保留舊聊天來源契約相容性，入口 readiness 仍失敗，因此不可視為完整 WP-11 部署。

原始 IP 用於業務存取判斷；native IPv6 /64 只用於限流 bucket。未知 IP 的聊天讀取／購買廣播在觸及限流器及業務資料庫前回應 503，不共用 unknown bucket。推薦入口 503 會產生固定 `platform_referral_source_unavailable` 事件，不附帶 request 資料；429 不產生該事件。

`cloudflare_waf` 沒有本次新增路由的遠端規則證據，遠端 memory 沒有跨实例共享額度，readiness 不將兩者視為本次可接受部署。Production 仍為未驗證／未授權狀態。

回滾僅回復該獨立 Preview 的精確 patch 或刪除本次建立的 Preview；不得修改既有 staging alias、使用者 dirty 工作樹或停用安全檢查來通過驗證。
