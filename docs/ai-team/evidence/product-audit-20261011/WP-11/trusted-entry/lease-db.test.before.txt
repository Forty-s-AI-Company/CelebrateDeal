import type { PrismaClient } from "@prisma/client";
import { randomUUID } from "node:crypto";
import { afterAll, afterEach, beforeEach, expect, it, vi } from "vitest";
import { getDb } from "@/lib/db";
import { POST, DELETE } from "./route";
import { resetInMemoryRateLimitForTests } from "@/lib/rate-limit";
import { admitLiveViewer, hashLiveViewerToken, LIVE_VIEWER_SESSION_TTL_MS } from "@/lib/live-quota-admission";

import { assertLocalTestDatabase } from "../../../../scripts/local-database-safety";
// 可在本機／CI disposable測試DB重跑，既有guard拒絕正式或互動式DB。
assertLocalTestDatabase("DATABASE_URL", process.env.DATABASE_URL);
const db = getDb();
const owned: string[] = [];
const now = new Date("2026-10-12T08:00:00.000Z");
beforeEach(() => { vi.stubEnv("NEXT_PUBLIC_APP_URL", "https://app.example.test"); vi.stubEnv("RATE_LIMIT_PROVIDER", "memory"); resetInMemoryRateLimitForTests(); vi.useFakeTimers({ toFake: ["Date"] }); vi.setSystemTime(now); });
afterEach(async () => { vi.useRealTimers(); resetInMemoryRateLimitForTests(); await db.vendor.deleteMany({ where: { id: { in: owned.splice(0) } } }); vi.unstubAllEnvs(); });
afterAll(() => db.$disconnect());
async function fixture() {
 const key = `wp11-lease-${randomUUID()}`;
 const vendor = await db.vendor.create({ data: { name: "Synthetic lease", slug: key, email: key + "@example.test", passwordHash: "synthetic-unused" } }); owned.push(vendor.id);
 const video = await db.video.create({ data: { vendorId: vendor.id, title: "Synthetic", sourceType: "url", videoUrl: "https://example.test/synthetic-unused.mp4", status: "ready" } });
 const form = await db.registrationForm.create({ data: { vendorId: vendor.id, slug: key, name: "Synthetic", headline: "Synthetic", fields: [{ key: "name", label: "Name", type: "text", required: true }, { key: "email", label: "Email", type: "email", required: true }] } });
 const template = await db.messageTemplate.create({ data: { vendorId: vendor.id, name: "Synthetic", subject: "Synthetic registration", body: "{{name}} {{unsubscribe_url}}" } });
 const live = await db.live.create({ data: { vendorId: vendor.id, videoId: video.id, formId: form.id, messageTemplateId: template.id, slug: key, title: "Synthetic", streamMode: "live", status: "live", scheduledAt: new Date(+now - 60_000), startedAt: new Date(+now - 60_000), quotaPolicy: { version: 1, affiliateMode: "enabled", maxConcurrentViewers: 1, stopWhenCreditsBelow: 0 } } });
 return { vendorId: vendor.id, liveId: live.id };
}
function request(method: string, payload: { vendorId: string; liveId: string }, value?: string) {
 return new Request("https://app.example.test/api/live-admission", { method, headers: { origin: "https://app.example.test", "content-type": "application/json", "x-celebratedeal-client": "web", "x-forwarded-for": "192.0.2.11", ...(value ? { cookie: `celebratedeal_live_viewer=${value}` } : {}) }, body: JSON.stringify(payload) });
}
it("rate-limited same-IP release stops counting at exactly the 90-second lease boundary", async () => {
 const payload = await fixture(); const admitted = await POST(request("POST", payload)); expect(admitted.status).toBe(200);
 const value = admitted.cookies.get("celebratedeal_live_viewer")!.value;
 for (let i = 0; i < 120; i++) expect((await DELETE(request("DELETE", payload))).status).toBe(200);
 expect((await DELETE(request("DELETE", payload, value))).status).toBe(429);
 expect(await db.liveViewerSession.count({ where: payload })).toBe(1);
 vi.setSystemTime(new Date(+now + LIVE_VIEWER_SESSION_TTL_MS - 1));
 expect((await POST(request("POST", payload))).status).toBe(429);
 vi.setSystemTime(new Date(+now + LIVE_VIEWER_SESSION_TTL_MS));
 expect((await POST(request("POST", payload))).status).toBe(200);
 expect(await db.liveViewerSession.count({ where: { ...payload, expiresAt: { gt: new Date() } } })).toBe(1);
});
it("an expired viewer cannot renew past a full viewer cap", async () => {
 const payload = await fixture(); const admitted = await POST(request("POST", payload)); expect(admitted.status).toBe(200);
 const value = admitted.cookies.get("celebratedeal_live_viewer")!.value;
 vi.setSystemTime(new Date(+now + LIVE_VIEWER_SESSION_TTL_MS));
 expect((await POST(request("POST", payload))).status).toBe(200);
 const before = await db.liveViewerSession.findUniqueOrThrow({ where: { tokenHash: hashLiveViewerToken(value) } });
 const rejected = await POST(request("POST", payload, value));
 expect(rejected.status).toBe(429);
 expect(rejected.cookies.get("celebratedeal_live_viewer")).toBeUndefined();
 expect((await db.liveViewerSession.findUniqueOrThrow({ where: { id: before.id } })).expiresAt).toEqual(before.expiresAt);
 expect(await db.liveViewerSession.count({ where: { ...payload, expiresAt: { gt: new Date() } } })).toBe(1);
});

it("valid renewal keeps its slot and an expired renewal can reclaim an empty slot", async () => {
 const payload = await fixture(); const initial = await POST(request("POST", payload)); expect(initial.status).toBe(200);
 const value = initial.cookies.get("celebratedeal_live_viewer")!.value;
 vi.setSystemTime(new Date(+now + LIVE_VIEWER_SESSION_TTL_MS - 1));
 expect((await POST(request("POST", payload, value))).status).toBe(200);
 expect(await db.liveViewerSession.count({ where: payload })).toBe(1);
 vi.setSystemTime(new Date(+now + 2 * LIVE_VIEWER_SESSION_TTL_MS));
 expect((await POST(request("POST", payload, value))).status).toBe(200);
 expect(await db.liveViewerSession.count({ where: payload })).toBe(1);
});
it("concurrent expired renewals cannot overfill the last slot", async () => {
 const payload = await fixture(); const a = await POST(request("POST", payload)); expect(a.status).toBe(200);
 const first = a.cookies.get("celebratedeal_live_viewer")!.value;
 vi.setSystemTime(new Date(+now + LIVE_VIEWER_SESSION_TTL_MS));
 const b = await POST(request("POST", payload)); expect(b.status).toBe(200);
 const second = b.cookies.get("celebratedeal_live_viewer")!.value;
 vi.setSystemTime(new Date(+now + 2 * LIVE_VIEWER_SESSION_TTL_MS));
 const results = await Promise.all([POST(request("POST", payload, first)), POST(request("POST", payload, second))]);
 expect(results.map(r => r.status).sort()).toEqual([200, 429]);
 expect(await db.liveViewerSession.count({ where: { ...payload, expiresAt: { gt: new Date() } } })).toBe(1);
});

it.each(["expired", "new"])("forces overlapping capacity reads for expired renewal versus %s", async (other) => {
 const payload = await fixture(); const first = await POST(request("POST", payload)); expect(first.status).toBe(200);
 const a = first.cookies.get("celebratedeal_live_viewer")!.value;
 vi.setSystemTime(new Date(+now + LIVE_VIEWER_SESSION_TTL_MS));
 let b: string | undefined;
 if (other === "expired") {
  const second = await POST(request("POST", payload)); expect(second.status).toBe(200);
  b = second.cookies.get("celebratedeal_live_viewer")!.value;
  vi.setSystemTime(new Date(+now + 2 * LIVE_VIEWER_SESSION_TTL_MS));
 }
 let reads = 0; let release = () => {};
 const barrier = new Promise<void>(resolve => { release = resolve; });
 // 真實count完成後才放行寫入，確保兩筆交易都讀到尚有空位。
 const client = db.$extends({ query: { liveViewerSession: { async count({ args, query }) {
  const count = await query(args); reads++;
  if (reads <= 2) { expect(count).toBe(0); if (reads === 2) release(); await barrier; }
  return count;
 } } } }) as unknown as PrismaClient;
 const results = await Promise.allSettled([admitLiveViewer(client, { ...payload, token: a }), admitLiveViewer(client, { ...payload, token: b })]);
 expect(results.filter(result => result.status === "fulfilled")).toHaveLength(1);
 const denied = results.find(result => result.status === "rejected");
 expect(denied?.status === "rejected" && denied.reason.code).toBe("viewer_limit_reached");
 expect(reads).toBeGreaterThanOrEqual(3);
 expect(await db.liveViewerSession.count({ where: { ...payload, expiresAt: { gt: new Date() } } })).toBe(1);
});
