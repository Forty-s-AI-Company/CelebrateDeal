"use client";
import { createContext, useContext, type ReactNode } from "react";
import type { RegistrationRetryDraft } from "@/lib/registration-form-retry";
import { formSubmissionErrorMessage } from "@/lib/registration-submission-client";
type RetryState = { formId?: string; sourcePath: string; draft: RegistrationRetryDraft | null; failed: boolean; submitted?: boolean };
const Context = createContext<RetryState | null>(null);
export function RegistrationRetryProvider({ value, children }: { value: RetryState; children: ReactNode }) {
  return <Context.Provider value={value}>{children}</Context.Provider>;
}
export function useRegistrationRetry(formId: string) {
  const state = useContext(Context);
  if (!state || state.formId !== formId) return null;
  // 網址可被分享或保留；沒有有效草稿時，不把網址旗標當成送出失敗的證據。
  return { ...state, message: state.failed ? state.draft ? formSubmissionErrorMessage(state.draft.status) : "目前沒有可恢復的填寫內容，無法確認先前的送出結果。若您已報名，請先查看確認信，避免重複送出；若尚未報名，請重新填寫。" : "" };
}

"use client";

import { useEffect, useRef } from "react";

/** Keep live regions mounted and restore keyboard position after a failed submit. */
export function RegistrationSubmissionFeedback({ pending, errorMessage = "", successMessage = "" }: {
  pending: boolean;
  errorMessage?: string;
  successMessage?: string;
}) {
  const errorRef = useRef<HTMLParagraphElement>(null);
  useEffect(() => {
    const alert = errorRef.current;
    if (!errorMessage || pending || !alert) return;
    const active = alert.ownerDocument.activeElement;
    const form = alert.closest("form");
    // 不搶走使用者等待期間移往其他欄位或連結的焦點。
    const lostFocus = !active || active === alert.ownerDocument.body || active === alert;
    const onSubmit = active instanceof HTMLButtonElement && active.type === "submit" && active.form === form;
    if (lostFocus || onSubmit) alert.focus();
  }, [errorMessage, pending]);

  return <>
    <p ref={errorRef} tabIndex={-1} role="alert" aria-live="assertive" aria-atomic="true" className={`text-sm text-red-700 ${pending || !errorMessage ? "sr-only" : ""}`}>{pending ? "" : errorMessage}</p>
    <p role="status" aria-live="polite" aria-atomic="true" className={pending || !successMessage || errorMessage ? "sr-only" : "text-sm text-emerald-700"}>{pending ? "正在送出，請勿重複操作。" : errorMessage ? "" : successMessage}</p>
  </>;
}

import { registrationResultCookieName, verifyRegistrationResult } from "@/lib/registration-submission-receipt";
import { cookies, headers } from "next/headers";
import type { ReactNode } from "react";
import { getDb } from "@/lib/db";
import { readRegistrationRetryDraft, registrationRetrySecureCookies, registrationRetryCookieName } from "@/lib/registration-form-retry";
import { RegistrationRetryProvider } from "@/components/registration-retry-context";

/** 公開入口載入後才讀草稿，避免跳過原本的發布與存取檢查。 */
export async function registrationRetryBoundary({ formId, sourcePath, retry, submitted }: {
  formId?: string; sourcePath: string; retry?: string | string[]; submitted?: string;
}) {
  const failed = typeof retry === "string" && ["400", "403", "404", "429", "503", "unavailable"].includes(retry);
  let draft = null;
  if (failed && retry !== "unavailable" && formId) {
    const jar = await cookies();
    const host = (await headers()).get("host") ?? "";
    const capability = jar.get(registrationRetryCookieName(registrationRetrySecureCookies(host)))?.value;
    try { draft = await readRegistrationRetryDraft(getDb(), { capability, formId, sourcePath }); } catch { /* 明示未保留，不送出敏感診斷。 */ }
  }
  const verifiedSubmission = await verifiedRegistrationSubmission(formId, sourcePath, submitted);
  return function RetryBoundary(children: ReactNode) {
    return <RegistrationRetryProvider value={{ formId, sourcePath, draft, failed, submitted: verifiedSubmission }}>{children}</RegistrationRetryProvider>;
  };
}


/** 共用驗證，避免個別入口直接信任網址上的成功旗標。 */
export async function verifiedRegistrationSubmission(formId: string | undefined, sourcePath: string, submitted?: string) {
  if (submitted !== "verification_required" || !formId) return false;
  const jar = await cookies();
  const secure = registrationRetrySecureCookies((await headers()).get("host") ?? "");
  return verifyRegistrationResult(jar.get(registrationResultCookieName(secure))?.value, formId, sourcePath);
}

export function LeadForm({
  formId,
  liveId,
  fields,
  sessions = [],
  submitLabel,
  successMessage,
  redirectTo,
  themeColor = "#c2410c",
}: {
  formId: string;
  liveId?: string | null;
  fields: FieldSpec[];
  sessions?: LeadFormSession[];
  submitLabel: string;
  successMessage: string;
  redirectTo?: string;
  themeColor?: string;
}) {
  const retry = useRegistrationRetry(formId);
  const [status, setStatus] = useState<"idle" | "loading" | "success" | "error">(retry?.submitted ? "success" : retry?.message ? "error" : "idle");
  const [errorMessage, setErrorMessage] = useState(retry?.message ?? "");
  const submissionInFlight = useRef(false);
  const [shareCode] = useState(() => typeof window === "undefined" ? "" : new URLSearchParams(window.location.search).get("share") ?? "");

  async function onSubmit(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();
    // React 尚未更新 disabled 前，也要擋住同一輪事件的重複送出。
    if (submissionInFlight.current) return;
    submissionInFlight.current = true;
    setStatus("loading");
    setErrorMessage("");
    const form = event.currentTarget;

    try {
      const formData = new FormData(form);
      const selectedLiveId = getSubmittedLiveId(formData, liveId);
      const payload = Object.fromEntries(
        [...formData.entries()].filter(([key]) => !["formId", "liveId", "referralCode", "shareCode", "redirectTo", "__registrationReturnPath"].includes(key)),
      );
      const referralCode = new URLSearchParams(window.location.search).get("ref");
      const response = await postRegistrationSubmission(buildFormSubmissionRequestBody({ formId, liveId: selectedLiveId, payload, referralCode, shareCode, landingPageId: new URLSearchParams(window.location.search).get("lp") }));

      if (response.ok) {
        setStatus("success");
        form.reset();
      } else {
        submissionInFlight.current = false;
        setStatus("error");
        setErrorMessage(formSubmissionErrorMessage(response.status));
      }
    } catch {
      submissionInFlight.current = false;
      setStatus("error");
      setErrorMessage(formSubmissionErrorMessage());
    }
  }

  if (status === "success") {
    return (
      <div role="status" aria-live="polite" className="grid gap-3">
        <p className="rounded-lg bg-emerald-50 p-4 text-sm font-medium text-emerald-800">{successMessage}</p>
        <p className="rounded-lg border border-emerald-100 bg-white p-4 text-sm leading-6 text-emerald-800">{FORM_SUBMISSION_VERIFICATION_MESSAGE}</p>

          🟢 接收 LINE 開播提醒與電子票券
        </LineLoginButton>
      </div>
    );
  }

  return (
    <form method="post" action="/api/form-submissions" onSubmit={onSubmit} aria-busy={status === "loading"} className="grid gap-4">
      <input type="hidden" name="formId" value={formId} />
      {retry ? <input type="hidden" name="__registrationReturnPath" value={retry.sourcePath} /> : null}
      {sessions.length === 0 && liveId ? <input type="hidden" name="liveId" value={liveId} /> : null}
      <input type="hidden" name="shareCode" value={shareCode} />
      {redirectTo || retry?.sourcePath ? <input type="hidden" name="redirectTo" value={redirectTo || retry?.sourcePath} /> : null}

      {sessions.length > 0 ? (
        <fieldset className="grid gap-3 rounded-xl border border-slate-200 bg-slate-50 p-4">
          <legend className="px-1 text-sm font-bold text-slate-900">選擇場次</legend>
          <div className="grid gap-2">
            {sessions.map((session, index) => (
              <label key={session.id} className="grid cursor-pointer gap-1 rounded-lg border border-slate-200 bg-white p-3 text-sm transition has-[:checked]:border-[var(--registration-form-theme)] has-[:checked]:ring-2 has-[:checked]:ring-blue-100" style={{ "--registration-form-theme": themeColor } as React.CSSProperties}>
                <span className="flex items-center gap-2 font-semibold text-slate-900">
                  <input type="radio" name="liveId" value={session.id} required defaultChecked={retry?.draft?.liveId && sessions.some((item) => item.id === retry.draft?.liveId) ? session.id === retry.draft.liveId : index === 0} className="h-4 w-4 accent-[var(--registration-form-theme)]" />
                  {session.title}

              </label>
            ))}
          </div>
        </fieldset>
      ) : null}

      {fields.map((field) => (
