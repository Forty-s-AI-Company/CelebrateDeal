/** Shared browser transport: keep attribution payloads intact and never expose server errors. */
export async function postRegistrationSubmission(body: Record<string, unknown>): Promise<Response> {
  const response = await fetch("/api/form-submissions", {
    method: "POST",
    redirect: "manual",
    headers: { "Content-Type": "application/json", "X-CelebrateDeal-Client": "web" },
    body: JSON.stringify(body),
  });
  if (!response.ok) return response;
  // 只接受 API 明確回覆；代理錯誤頁或重新導向不能冒充報名成功。
  try {
    if (!response.redirected && /^application\/json(?:\s*;|$)/iu.test(response.headers.get("content-type") ?? "")) {
      const result: unknown = await response.clone().json();
      if (result && typeof result === "object" && "ok" in result && result.ok === true && "verificationRequired" in result && result.verificationRequired === true) return response;
    }
  } catch { /* 回應格式不符時保留欄位，不向使用者顯示伺服器內容。 */ }
  return new Response(null, { status: 503 });
}

export function formSubmissionErrorMessage(status?: number) {
  if (status === 400) return "請確認必填欄位與資料格式後再送出。";
  if (status === 403) return "這次送出未通過安全檢查，請重新整理頁面後再試。";
  if (status === 429) return "送出次數太頻繁，請稍候一分鐘再試。";
  if (status === 404 || status === 503) return "這張表單目前無法接收資料，請稍後再試。";
  // 已收到 HTTP 回應時，不把伺服器拒絕或故障誤指為網路問題。
  if (status !== undefined) return "資料暫時無法送出；內容仍保留，請稍後再試。";
  return "送出失敗；內容仍保留，請檢查網路後再試。";
}

  const inFlight = useRef(false);
  useEffect(() => {
    // 從瀏覽器歷史快取返回時，允許使用者重新送出。
    const reset = () => { inFlight.current = false; setPending(false); };

  function onSubmit(event: React.FormEvent<HTMLFormElement>) {
    if (preview || inFlight.current) { event.preventDefault(); return; }
    inFlight.current = true;

    flushSync(() => setPending(true));

  }

"use client";

import { useEffect, useRef } from "react";

/** Keep live regions mounted and restore keyboard position after a failed submit. */
export function RegistrationSubmissionFeedback({ pending, errorMessage = "", successMessage = "", focusSuccess = false }: {
  pending: boolean;
  errorMessage?: string;
  successMessage?: string;
  focusSuccess?: boolean;
}) {
  const successRef = useRef<HTMLParagraphElement>(null);
  useEffect(() => {
    if (focusSuccess && successMessage && !pending && !errorMessage) successRef.current?.focus();
  }, [focusSuccess, successMessage, pending, errorMessage]);
  const errorRef = useRef<HTMLParagraphElement>(null);
  useEffect(() => {
    const alert = errorRef.current;
    if (!errorMessage || pending || !alert) return;
    const active = alert.ownerDocument.activeElement;
    const form = alert.closest("form");
    // 不搶走使用者等待期間移往其他欄位或連結的焦點。
    const lostFocus = !active || active === alert.ownerDocument.body || active === alert;
    const onSubmit = active instanceof HTMLButtonElement && active.type === "submit" && active.form === form;
    if (lostFocus || onSubmit) alert.focus();
  }, [errorMessage, pending]);

  return <>
    <p ref={errorRef} tabIndex={-1} role="alert" aria-live="assertive" aria-atomic="true" className={`text-sm text-red-700 ${pending || !errorMessage ? "sr-only" : ""}`}>{pending ? "" : errorMessage}</p>
    <p ref={successRef} tabIndex={focusSuccess ? -1 : undefined} role="status" aria-live="polite" aria-atomic="true" className={pending || !successMessage || errorMessage ? "sr-only" : "text-sm text-emerald-700"}>{pending ? "正在送出，請勿重複操作。" : errorMessage ? "" : successMessage}</p>
  </>;
}
