"use client";

import { useRegistrationRetry } from "@/components/registration-retry-context";
import { FORM_SUBMISSION_VERIFICATION_MESSAGE } from "@/lib/registration-submission-client";

/** 截止頁只提供原來源綁定的結果及唯讀草稿，不提供任何送出入口。 */
export function RegistrationClosedFeedback({ formId, fields = [], destination }: {
  formId?: string;
  fields?: Array<{ key: string; label: string }>;
  destination?: string;
}) {
  const retry = useRegistrationRetry(formId ?? "");
  return <main className="mx-auto grid max-w-2xl gap-4 px-6 py-16">
    <h1 className="text-2xl font-bold">此活動已截止</h1>
    <p>目前無法再次送出報名。</p>
    {retry?.submitted ? <p role="status">{FORM_SUBMISSION_VERIFICATION_MESSAGE}</p>
      : <p role="alert">{retry?.draft ? "先前送出未完成。以下保留當時填寫的內容，僅供查看。" : "無法確認先前的送出結果，也沒有可恢復的填寫內容。若您已報名，請查看確認信。"}</p>}
    {!retry?.submitted && retry?.draft ? <div className="grid gap-3">
      {fields.filter((field) => Object.hasOwn(retry.draft!.answers, field.key)).map((field) => <label key={field.key} className="grid gap-1 text-sm">
        {field.label}<input readOnly value={retry.draft!.answers[field.key]} className="min-h-11 rounded border bg-slate-50 px-3" />
      </label>)}
    </div> : null}
    {destination ? <a href={destination} className="underline">前往活動指定頁面</a> : null}
  </main>;
}

import type { ReactNode } from "react";
import { redirect } from "next/navigation";
import type { FunnelRuntimeDecision } from "@/lib/funnel-runtime";
import { RegistrationClosedFeedback } from "./registration-closed-feedback";
import { hasRegistrationFeedbackQuery } from "./registration-retry-boundary";

/** 截止規則先決定可用性；只有結果查詢才留在原來源提供唯讀內容。 */
export function registrationDeadlineOutcome({ decision, query, form, slug, withRetry }: {
  decision: FunnelRuntimeDecision;
  query: { registrationRetry?: string; submitted?: string };
  form?: { id: string; fields: Array<{ key: string; label: string }> };
  slug: string;
  withRetry: (children: ReactNode) => ReactNode;
}) {
  if (decision.status !== "closed" && decision.status !== "redirect") return null;
  const destination = decision.status === "redirect" ? `/lp/${encodeURIComponent(slug)}/${encodeURIComponent(decision.path)}` : undefined;
  if (hasRegistrationFeedbackQuery(query)) return withRetry(<RegistrationClosedFeedback formId={form?.id} fields={form?.fields} destination={destination} />);
  if (destination) redirect(destination);
  return <main className="mx-auto max-w-2xl px-6 py-24 text-center"><h1 className="text-2xl font-black text-slate-950">此活動已截止</h1><p className="mt-3 text-slate-600">目前不再接受新的瀏覽或送出資料。</p></main>;
}

import { registrationResultCookieName, verifyRegistrationResult } from "@/lib/registration-submission-receipt";
import { cookies, headers } from "next/headers";
import type { ReactNode } from "react";
import { getDb } from "@/lib/db";
import { readRegistrationRetryDraft, registrationRetrySecureCookies, registrationRetryCookieName } from "@/lib/registration-form-retry";
import { RegistrationRetryProvider } from "@/components/registration-retry-context";

/** 公開入口載入後才讀草稿，避免跳過原本的發布與存取檢查。 */
export async function registrationRetryBoundary({ formId, sourcePath, retry, submitted }: {
  formId?: string; sourcePath: string; retry?: string | string[]; submitted?: string;
}) {
  const failed = typeof retry === "string" && ["400", "403", "404", "429", "503", "unavailable"].includes(retry);
  let draft = null;
  if (failed && retry !== "unavailable" && formId) {
    const jar = await cookies();
    const host = (await headers()).get("host") ?? "";
    const capability = jar.get(registrationRetryCookieName(registrationRetrySecureCookies(host)))?.value;
    try { draft = await readRegistrationRetryDraft(getDb(), { capability, formId, sourcePath }); } catch { /* 明示未保留，不送出敏感診斷。 */ }
  }
  const verifiedSubmission = await verifiedRegistrationSubmission(formId, sourcePath, submitted);
  return function RetryBoundary(children: ReactNode) {
    return <RegistrationRetryProvider value={{ formId, sourcePath, draft, failed, submitted: verifiedSubmission }}>{children}</RegistrationRetryProvider>;
  };
}


/** 共用驗證，避免個別入口直接信任網址上的成功旗標。 */
export async function verifiedRegistrationSubmission(formId: string | undefined, sourcePath: string, submitted?: string) {
  if (submitted !== "verification_required" || !formId) return false;
  const jar = await cookies();
  const secure = registrationRetrySecureCookies((await headers()).get("host") ?? "");
  return verifyRegistrationResult(jar.get(registrationResultCookieName(secure))?.value, formId, sourcePath);
}

/** 網址只決定是否呈現結果頁；成功與欄位仍須通過原來源綁定驗證。 */
export function hasRegistrationFeedbackQuery(query: { registrationRetry?: string; submitted?: string }) {
  return ["400", "403", "404", "429", "503", "unavailable"].includes(query.registrationRetry ?? "") || query.submitted === "verification_required";
}

"use client";
import { createContext, useContext, type ReactNode } from "react";
import type { RegistrationRetryDraft } from "@/lib/registration-form-retry";
import { formSubmissionErrorMessage } from "@/lib/registration-submission-client";
type RetryState = { formId?: string; sourcePath: string; draft: RegistrationRetryDraft | null; failed: boolean; submitted?: boolean };
const Context = createContext<RetryState | null>(null);
export function RegistrationRetryProvider({ value, children }: { value: RetryState; children: ReactNode }) {
  return <Context.Provider value={value}>{children}</Context.Provider>;
}
export function useRegistrationRetry(formId: string) {
  const state = useContext(Context);
  if (!state || state.formId !== formId) return null;
  // 網址可被分享或保留；沒有有效草稿時，不把網址旗標當成送出失敗的證據。
  return { ...state, message: state.failed ? state.draft ? formSubmissionErrorMessage(state.draft.status) : "目前沒有可恢復的填寫內容，無法確認先前的送出結果。若您已報名，請先查看確認信，避免重複送出；若尚未報名，請重新填寫。" : "" };
}
