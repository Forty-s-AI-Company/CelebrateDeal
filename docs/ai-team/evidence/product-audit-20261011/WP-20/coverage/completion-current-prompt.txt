const id = z.string().regex(/^[A-Za-z0-9_-]{1,128}$/);
const leaseSchema = z.object({ vendorId: id, liveId: id, versionId: id, reportId: id, workerId: id }).strict();
const factSchema = z.object({ personId: id, firstCompletedAtMs: z.number().int().min(0).max(8_640_000_000_000_000) }).strict();
const factsSchema = z.array(factSchema).max(1_000_000).superRefine((rows, context) => {
  if (new Set(rows.map(row => row.personId)).size !== rows.length) context.addIssue({ code: "custom", message: "Duplicate completion person" });
});
type Database = Pick<Prisma.TransactionClient, "$queryRaw">;
export type StoredCompletion = z.infer<typeof factSchema>;

/** Internal writer: caller must establish complete registration/viewing coverage
 * and delivery before calculating candidates. The DB rechecks lease/revision;
 * a worker result alone is not permission to certify missing source history.
 */
export async function publishMetricCompletion(db: Pick<PrismaClient, "$transaction">, value: MetricReportLease, sourceRevision: bigint,
  scope: MetricScope, projection: CompletionProjection): Promise<boolean> {
  const lease = leaseSchema.parse(value);
  if (typeof sourceRevision !== "bigint" || sourceRevision < BigInt(0)) throw new Error("Invalid completion source revision");
  if (scope.vendorId !== lease.vendorId || scope.liveId !== lease.liveId || scope.contentVersionId !== lease.versionId
    || !["live", "replay"].includes(scope.mode)) throw new Error("Completion scope mismatch");
  if (projection.status !== "available") return false;
  const cohortCount = z.number().int().min(0).max(1_000_000).parse(projection.cohortCount);
  if (projection.rows.some(row => row.scope.vendorId !== scope.vendorId || row.scope.liveId !== scope.liveId
    || row.scope.contentVersionId !== scope.contentVersionId || row.scope.mode !== scope.mode)) throw new Error("Completion scope mismatch");
  const rows = factsSchema.parse(projection.rows.map(row => ({ personId: row.personId, firstCompletedAtMs: row.firstCompletedAtMs })));
  if (rows.length > cohortCount) throw new Error("Completion cohort mismatch");
  // Serialize before opening the transaction so payload conversion holds no locks.
  const payload = JSON.stringify(rows);
  return db.$transaction(async tx => {
    // SET LOCAL applies before the publish statement starts and expires on rollback.
    await tx.$queryRaw(Prisma.sql`SELECT set_config('statement_timeout','5000',true), set_config('lock_timeout','2000',true)`);
    const [result] = await tx.$queryRaw<{ changed: number }[]>(Prisma.sql`
      SELECT public.wp20_metric_completion_publish(${lease.vendorId},${lease.liveId},${lease.versionId},${lease.reportId},${lease.workerId},
        ${sourceRevision}::bigint,${cohortCount}::integer,${payload}::jsonb) AS changed`);
    return result?.changed === 1;
  }, { isolationLevel: Prisma.TransactionIsolationLevel.ReadCommitted, timeout: 10000, maxWait: 5000 });
}

/** Read in the same RR snapshot as viewing facts. null is missing provenance;
 * [] is a stored complete empty projection, never inferred from a missing row.
 */
export async function readMetricCompletion(tx: Database, value: MetricReportLease, reserve: (rows: number) => boolean = () => true) {
  const lease = leaseSchema.parse(value);
  const [record] = await tx.$queryRaw<{ isolation: string; rowCount: number | null; cohortCount: number | null }[]>(Prisma.sql`
    SELECT current_setting('transaction_isolation') AS isolation,p."rowCount",p."cohortCount"
    FROM "LiveMetricReport" r JOIN "LiveMetricVersion" v ON v.id=r."versionId" AND v."vendorId"=r."vendorId" AND v."liveId"=r."liveId"
    LEFT JOIN "LiveMetricCompletionProjection" p ON p."reportId"=r.id AND p."sourceRevision"=r."inputRevision" AND p."algorithmVersion"=1
    WHERE r.id=${lease.reportId} AND r."vendorId"=${lease.vendorId} AND r."liveId"=${lease.liveId} AND r."versionId"=${lease.versionId}
      AND r.status='running' AND r."leaseOwner"=${lease.workerId} AND r."leaseExpiresAt">clock_timestamp() AND r."expiresAt">clock_timestamp()
      AND r."inputRevision"=v."sourceRevision" AND r."inputOrderRevision"=public.wp20_metric_order_revision(r."vendorId",r."liveId")`);
  if (!record) return { status: "unavailable" as const, reason: "lease_or_scope_unavailable" as const };
  if (record.isolation !== "repeatable read") throw new Error("Completion reader requires repeatable-read snapshot");
  // Reserve the shared budget before transferring or parsing the JSON payload.
  const rowCount = record.rowCount === null ? 0 : z.number().int().min(0).max(1_000_000).parse(record.rowCount);
  if (!reserve(rowCount + 1)) return { status: "unavailable" as const, reason: "limit_exceeded" as const };
  if (record.cohortCount === null) return { status: "available" as const, rows: null, cohortCount: null };
  const cohortCount = z.number().int().min(0).max(1_000_000).parse(record.cohortCount);
  const [payload] = await tx.$queryRaw<{ rows: unknown }[]>(Prisma.sql`
    SELECT rows FROM "LiveMetricCompletionProjection" WHERE "reportId"=${lease.reportId}`);
  const rows = factsSchema.parse(payload?.rows);
  if (rows.length !== rowCount) throw new Error("Completion row count mismatch");
  if (rows.length > cohortCount) throw new Error("Completion cohort mismatch");
  return { status: "available" as const, rows, cohortCount };
}

import { claimMetricReport, finishMetricReport, type MetricReportLease } from "./live-metric-report-lease";

/** Internal single-job orchestration; caller authenticates and chooses a scoped
 * queued report. Supply the bounded worker executor, never a browser callback.
 * No automatic retries or synchronous computation fallback. Unfinished leases
 * expire through the existing DB-clock protocol so later attempts can recover.
 */
export async function runMetricReportJob(db: Database, value: MetricReportLease, execute: Executor, project?: ReturnType<typeof createCompletionExecutor>) {
  const lease = { ...value };
  let stage: "claim" | "snapshot" | "conversion" | "execution" | "projection" | "publish" | "reload" | "finish" = "claim";
  try {
    if (!await claimMetricReport(db, lease)) return { status: "not_claimed" as const };
    stage = "snapshot";
    let snapshot = await loadMetricViewingSnapshot(db, lease);
    if (snapshot.status !== "available") return { status: "snapshot_unavailable" as const, reason: snapshot.reason };
    stage = "conversion";
    let input = metricSnapshotInput(snapshot.data);
    if (project && input.completions === null && input.registrations !== null && input.viewing !== null && input.durationMs !== null) {
      const sourceRevision = snapshot.data.context.sourceRevision;
      if (typeof sourceRevision !== "bigint") throw new Error("Missing snapshot revision");
      stage = "projection";
      const candidate = await project(input);
      if (candidate.status !== "available") return { status: "execution_unavailable" as const, reason: candidate.reason };
      if (candidate.projection.status === "available") {
        stage = "publish";
        if (!await publishMetricCompletion(db, lease, sourceRevision, input.scope, candidate.projection)) return { status: "lease_lost" as const };
        stage = "reload";
        snapshot = await loadMetricViewingSnapshot(db, lease);
        if (snapshot.status !== "available") return { status: "snapshot_unavailable" as const, reason: snapshot.reason };
        input = metricSnapshotInput(snapshot.data);
        if (input.completions === null || snapshot.data.context.sourceRevision !== sourceRevision) return { status: "lease_lost" as const };
      }
    }
    stage = "execution";
    const result = await execute(input);
    if (result.status !== "available") return { status: "execution_unavailable" as const, reason: result.reason };
    stage = "finish";
    // Finish revalidates the lease and source revision after expensive work.
    // Only aggregate output is persisted; raw person/browser facts stay private.
    if (!await finishMetricReport(db, lease, result.metrics)) return { status: "lease_lost" as const };
    return { status: "recorded" as const };
  } catch {
    // Failure summaries carry no database errors, identities or worker payloads.
    return { status: "failed" as const, stage };
  }
}

        if (!page.hasMore) break;
        afterLoss = page.next;
      }
      // A proven gap is decisive even while the collection-start audit is unknown.
      if (result.visitLosses.length) result.visitCoverage = { status: "unavailable", reason: "evidence_gap" };
      let after: { id: string; visitedAt: Date } | null = null;
      for (;;) {
        if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
        const page = await readMetricVisitPage(tx, { ...lease, after, take: 500 });
        if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
        if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");
        result.visits.push(...page.rows);
        if (!page.hasMore) break;
        after = page.next;
      }
      let afterId: string | null = null;
      for (;;) {
        if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
        const page = await readMetricRegistrationPage(tx, { ...lease, afterId, take: 100 });
        if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
        if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");
        for (const registration of page.rows) {
          if (!registration.identityCurrent) throw new SnapshotUnavailable("identity_changed");
          result.registrations.push(registration);
          await appendRegistrationSessions(tx, lease, registration.id, budget, result);
        }
        if (!page.hasMore) break;
        afterId = page.nextId;
      }
      let afterOrder: string | null = null;
      for (;;) {
        if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
        const page = await readMetricOrderPage(tx, { ...lease, afterId: afterOrder, take: 500 });
        if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
        if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");
        result.orders.push(...page.rows);
        if (!page.hasMore) {
          result.completionProjection = await loadCompletion(tx, lease, budget);
          return result;
        }
        afterOrder = page.nextId;
      }
    }), { isolationLevel: Prisma.TransactionIsolationLevel.RepeatableRead, timeout: 30000, maxWait: 5000 });
    return { status: "available", data };
  } catch (error) {
    if (error instanceof SnapshotUnavailable) return { status: "unavailable", reason: error.reason };
    // Infrastructure failures propagate to the job runner. They are never
    // converted into an empty cohort or a successfully calculated zero.
    throw error;
  }
}

async function appendRegistrationSessions(tx: Pick<Prisma.TransactionClient, "$queryRaw">, lease: MetricReportLease, registrationId: string,
  budget: MetricReadBudget, result: MetricViewingSnapshot) {

  for (;;) {
    if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
    const page = await readMetricSessionPage(tx, { ...lease, registrationId, afterId, take: 100 });
    if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
    if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");

      result.sessions.push({ registrationId, session, intervals: intervals.rows });
      result.deliveryComplete &&= intervals.deliveryComplete;
    }
    if (!page.hasMore) return;
    afterId = page.nextId;
  }
}

}

export type MetricExecution = { status: "available"; metrics: CoreMetrics }
  | { status: "unavailable"; reason: "busy" | "input_limit" | "timeout" | "worker_failed" };

/** The job runner supplies the trusted build artifact, never a request URL.
 * One calculation per executor; do not queue unbounded payloads in memory.
 * No synchronous fallback: worker failure must not block the web event loop.
 */
export function createMetricExecutor(artifact: URL, timeoutMs = 20000) {

  return async (input: CoreMetricsInput): Promise<MetricExecution> => {
    const result = await execute(input);
    return result.status === "available" ? { status: "available", metrics: result.value } : result;
  };
}
export type CompletionExecution = { status: "available"; projection: CompletionProjection }
  | { status: "unavailable"; reason: "busy" | "input_limit" | "timeout" | "worker_failed" };

/** Projection remains a candidate until a revision-checked atomic DB write. */
export function createCompletionExecutor(artifact: URL, timeoutMs = 20000) {
  const execute = createBoundedExecutor<CompletionProjection>(artifact, "completion", timeoutMs);
  return async (input: CompletionProjectionInput): Promise<CompletionExecution> => {
    const result = await execute({ ...input, visits: null, completions: null, orders: null });
    return result.status === "available" ? { status: "available", projection: result.value } : result;
  };
}
type BoundedResult<T> = { status: "available"; value: T }
  | { status: "unavailable"; reason: "busy" | "input_limit" | "timeout" | "worker_failed" };
function createBoundedExecutor<T>(artifact: URL, operation: "metrics" | "completion", timeoutMs: number) {
  if (artifact.protocol !== "file:" || !Number.isInteger(timeoutMs) || timeoutMs < 1 || timeoutMs > 30000) throw new Error("Invalid metric executor configuration");
  let active = false;
  return async (input: CoreMetricsInput): Promise<BoundedResult<T>> => {
    if (active) return { status: "unavailable", reason: "busy" };
    const lists = [input.visits, input.registrations, input.viewing, input.completions, input.orders];
    if (lists.some(rows => rows !== null && !Array.isArray(rows))
      || lists.reduce((sum, rows) => sum + (rows?.length ?? 0), 0) > 1_000_000) return { status: "unavailable", reason: "input_limit" };
    active = true;
    try {
      return await new Promise<BoundedResult<T>>(resolve => {
        let worker: Worker;
        try {
          worker = new Worker(artifact, { workerData: { operation, input }, env: {}, execArgv: [], stdout: true, stderr: true,
            resourceLimits: { maxOldGenerationSizeMb: 256, maxYoungGenerationSizeMb: 32, stackSizeMb: 4 } });
        } catch { resolve({ status: "unavailable", reason: "worker_failed" }); return; }
        // Consume and discard worker diagnostics; raw facts never enter logs.
        worker.stdout?.resume(); worker.stderr?.resume();
        let settled = false;
        const finish = (result: BoundedResult<T>) => {
          if (settled) return;
          settled = true; clearTimeout(timer);
          void worker.terminate().catch(() => undefined).then(() => resolve(result));
        };
        const timer = setTimeout(() => finish({ status: "unavailable", reason: "timeout" }), timeoutMs);
        worker.once("message", (value: T) => finish({ status: "available", value }));
        worker.once("error", () => finish({ status: "unavailable", reason: "worker_failed" }));
        worker.once("exit", () => finish({ status: "unavailable", reason: "worker_failed" }));
      });
    } finally { active = false; }
  };
}

import { evaluatePostCompletionOrder, POST_COMPLETION_ORDER_WINDOW_MS } from "./live-metric-contract";
import { firstCompletionTime } from "./live-viewing-evidence";

// A normal 100-viewer, one-hour event at five-second samples has 72,000
// distinct rows. Keep that supported; duplicates never consume this budget.
export const MAX_REPORT_COMPLETION_SEGMENTS = 1000000;

/** Pure reporting contract. The persistence adapter must provide complete,
 * verified facts for one tenant/live/content version and viewing mode.
 * null means unknown history; [] means an audited, complete empty history.
 * IDs are opaque internal references, never email addresses or session counts.
 */
export type MetricScope = Readonly<{
  vendorId: string;
  liveId: string;
  contentVersionId: string;
  mode: "live" | "replay";
}>;
type Scoped = Readonly<{ scope: MetricScope }>;
export type VisitFact = Scoped & Readonly<{ browserId: string; visitedAtMs: number }>;
export type RegistrationFact = Scoped & Readonly<{
  personId: string; browserId: string | null; verifiedAtMs: number;
}>;
/** Each segment has a constant playback rate; seek/pause/buffering split it.
 * Wall timestamps are validated by ingestion, not taken on trust from clients.
 */
export type ViewingFact = Scoped & Readonly<{
  personId: string;
  wallStartMs: number; wallEndMs: number;
  mediaStartMs: number; mediaEndMs: number;
}>;
/** Durable first crossing uses verified-eligible exposure only. A crossing
 * before verification contradicts this contract and invalidates the batch. */
export type CompletionFact = Scoped & Readonly<{ personId: string; firstCompletedAtMs: number }>;
/** Rows are current order snapshots from a read transaction, never historical
 * order events; the adapter must omit stale pre-cancellation states. */
export type OrderFact = Scoped & Readonly<{
  personId: string; establishedAtMs: number; cancelled: boolean;
}>;
export type CoreMetricsInput = Readonly<{
  scope: MetricScope;
  startMs: number; endMs: number; durationMs: number | null;
  visits: readonly VisitFact[] | null;
  registrations: readonly RegistrationFact[] | null;
  viewing: readonly ViewingFact[] | null;
  completions: readonly CompletionFact[] | null;
  orders: readonly OrderFact[] | null;
}>;
export type MetricValue =
  | { status: "unavailable"; reason: "missing_data" | "invalid_data" | "limit_exceeded" }
  | { status: "not_applicable"; reason: "zero_denominator" }
  | { status: "available"; value: number; numerator: number; denominator: number; unit: "percent" | "seconds" };
export type CoreMetrics = {
  registrationRate: MetricValue;
  attendanceRate: MetricValue;
  completionRate: MetricValue;
  averageViewingSeconds: MetricValue;
  postCompletionOrderRate: MetricValue;
  unattributedRegistrations: number | null;
  provisionalConversions: boolean | null;
};

const missing = (): MetricValue => ({ status: "unavailable", reason: "missing_data" });
const invalid = (): MetricValue => ({ status: "unavailable", reason: "invalid_data" });
const timestamp = (value: number) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000;
const id = (value: string) => typeof value === "string" && value.length > 0;
const scopeValid = (scope: MetricScope) => scope && id(scope.vendorId) && id(scope.liveId) && id(scope.contentVersionId) && (scope.mode === "live" || scope.mode === "replay");
const sameScope = (a: MetricScope, b: MetricScope) => scopeValid(a) && a.vendorId === b.vendorId && a.liveId === b.liveId && a.contentVersionId === b.contentVersionId && a.mode === b.mode;

function ratio(numerator: number, denominator: number, unit: "percent" | "seconds" = "percent"): MetricValue {
  if (!denominator) return { status: "not_applicable", reason: "zero_denominator" };
  const value = numerator / denominator * (unit === "percent" ? 100 : 1);
  return Number.isFinite(value) ? { status: "available", value, numerator, denominator, unit } : invalid();
}

/** Union actual exposure time across sessions/devices. Never fill gaps. */
function unionLength(intervals: readonly (readonly [number, number])[]): number {
  let end = 0;
  let total = 0;
  for (const [start, nextEnd] of [...intervals].sort((a, b) => a[0] - b[0])) {
    total += Math.max(0, nextEnd - Math.max(end, start));
    end = Math.max(end, nextEnd);
  }
  return total;
}

class InvalidMetricFacts extends Error {}
const requireFact = (condition: unknown) => { if (!condition) throw new InvalidMetricFacts(); };
type Cohort = Map<string, RegistrationFact>;

function validateScopeAndWindow(input: CoreMetricsInput) {
  requireFact(input && scopeValid(input.scope) && timestamp(input.startMs) && timestamp(input.endMs) && input.startMs < input.endMs);
  // Fail closed on mixed scopes; silently dropping rows would change denominators.
  for (const rows of [input.visits, input.registrations, input.viewing, input.completions, input.orders]) {
    requireFact(rows === null || (Array.isArray(rows) && Array.from(rows).every((row) => row && sameScope(row.scope, input.scope))));
  }
}

function buildCohort(input: CoreMetricsInput): Cohort {
  const registrations: Cohort = new Map();
  for (const row of input.registrations!) {
    requireFact(id(row.personId) && timestamp(row.verifiedAtMs) && (row.browserId === null || id(row.browserId)));
    const previous = registrations.get(row.personId);
    requireFact(!previous || (previous.verifiedAtMs === row.verifiedAtMs && previous.browserId === row.browserId));
    registrations.set(row.personId, row);
  }
  return new Map([...registrations].filter(([, row]) => row.verifiedAtMs >= input.startMs && row.verifiedAtMs < input.endMs));
}

function registrationMetrics(input: CoreMetricsInput, cohort: Cohort, result: CoreMetrics) {
  if (input.visits === null) return;
  const visits = new Map<string, number>();
  for (const row of input.visits) {
    requireFact(id(row.browserId) && timestamp(row.visitedAtMs));
    if (row.visitedAtMs >= input.startMs && row.visitedAtMs < input.endMs) visits.set(row.browserId, Math.min(visits.get(row.browserId) ?? Infinity, row.visitedAtMs));
  }
  const converted = new Set<string>();
  let unattributed = 0;
  for (const row of cohort.values()) {
    const visit = row.browserId === null ? undefined : visits.get(row.browserId);
    if (visit !== undefined && row.verifiedAtMs >= visit) converted.add(row.browserId!);
    else unattributed++;
  }
  result.registrationRate = ratio(converted.size, visits.size);
  result.unattributedRegistrations = unattributed;
}

function collectSegments(input: CoreMetricsInput, cohort: Cohort): Map<string, ViewingFact[]> {
  const segments = new Map<string, ViewingFact[]>();
  for (const row of input.viewing!) {
    requireFact(id(row.personId) && timestamp(row.wallStartMs) && timestamp(row.wallEndMs) && row.wallEndMs > row.wallStartMs);
    requireFact(Number.isSafeInteger(row.mediaStartMs) && Number.isSafeInteger(row.mediaEndMs) && row.mediaStartMs >= 0 && row.mediaEndMs > row.mediaStartMs);
    if (input.durationMs !== null && input.durationMs > 0) requireFact(row.mediaEndMs <= input.durationMs);
    const registration = cohort.get(row.personId);
    if (!registration || row.wallEndMs <= registration.verifiedAtMs || row.wallStartMs >= input.endMs) continue;
    const rows = segments.get(row.personId) ?? [];
    rows.push(row);
    segments.set(row.personId, rows);
  }
  // Deduplicate each person's references in place instead of retaining a second
  // string representation of every interval across the whole report.
  const compare = (a: ViewingFact, b: ViewingFact) => a.wallStartMs - b.wallStartMs
    || a.wallEndMs - b.wallEndMs || a.mediaStartMs - b.mediaStartMs || a.mediaEndMs - b.mediaEndMs;
  for (const rows of segments.values()) {
    rows.sort(compare);
    let retained = 0;
    for (const row of rows) {
      if (retained === 0 || compare(rows[retained - 1], row) !== 0) rows[retained++] = row;
    }
    rows.length = retained;
  }
  return segments;
}

function viewingMetrics(input: CoreMetricsInput, cohort: Cohort, result: CoreMetrics): Map<string, number> {
  const segments = collectSegments(input, cohort);
  const completed = new Map<string, number>();
  let completionLimit = [...segments.values()].reduce((count, rows) => count + rows.length, 0) > MAX_REPORT_COMPLETION_SEGMENTS;
  let wallMs = 0;
  for (const [person, rows] of segments) {
    const verifiedAt = cohort.get(person)!.verifiedAtMs;
    wallMs += unionLength(rows.map((row) => [Math.max(row.wallStartMs, verifiedAt, input.startMs), Math.min(row.wallEndMs, input.endMs)] as const));
    if (input.durationMs === null || completionLimit) continue;
    const completion = firstCompletionTime(input.durationMs, rows, { startMs: Math.max(verifiedAt, input.startMs), endMs: input.endMs - 1 });
    if (completion.status === "unavailable" && completion.reason === "limit_exceeded") {
      completionLimit = true;
      continue;
    }
    requireFact(completion.status === "available");
    if (completion.status === "available" && completion.firstCompletedAtMs !== null) completed.set(person, completion.firstCompletedAtMs);
  }
  result.attendanceRate = ratio(segments.size, cohort.size);
  result.averageViewingSeconds = ratio(wallMs / 1000, segments.size, "seconds");
  result.completionRate = input.durationMs === null ? missing() : completionLimit ? { status: "unavailable", reason: "limit_exceeded" } : ratio(completed.size, segments.size);
  return completed;
}

function orderMetrics(input: CoreMetricsInput, cohort: Cohort, completed: Map<string, number>, result: CoreMetrics) {
  if (result.completionRate.status === "unavailable") {
    result.postCompletionOrderRate = { ...result.completionRate };
    return;
  }
  if (input.completions === null || input.orders === null || input.durationMs === null) return;
  const firstCompletion = new Map<string, number>();
  for (const row of input.completions) {
    requireFact(id(row.personId) && timestamp(row.firstCompletedAtMs));
    const previous = firstCompletion.get(row.personId);
    requireFact(previous === undefined || previous === row.firstCompletedAtMs);
    firstCompletion.set(row.personId, row.firstCompletedAtMs);
  }
  for (const row of input.orders) requireFact(id(row.personId) && timestamp(row.establishedAtMs) && typeof row.cancelled === "boolean");
  // Reconcile durable first-crossing facts with independently derived viewing
  // evidence. [] is known-empty, so a contradiction is invalid, not missing.
  for (const [person, derivedAt] of completed) {
    requireFact(firstCompletion.get(person) === derivedAt);
  }
  for (const [person, at] of firstCompletion) {
    if (!cohort.has(person) || at >= input.endMs) continue;
    requireFact(at >= cohort.get(person)!.verifiedAtMs && completed.get(person) === at);
  }
  const ordered = new Set<string>();
  for (const row of input.orders) {
    if (!completed.has(row.personId) || row.establishedAtMs >= input.endMs) continue;
    const eligibility = evaluatePostCompletionOrder(firstCompletion.get(row.personId)!, row);
    if (eligibility.status === "available" && eligibility.qualifies) ordered.add(row.personId);
  }
  result.postCompletionOrderRate = ratio(ordered.size, completed.size);
  // The last millisecond of the inclusive seven-day window must be observed.
  result.provisionalConversions = [...completed.keys()].some((person) => input.endMs <= firstCompletion.get(person)! + POST_COMPLETION_ORDER_WINDOW_MS);
}

export function calculateCoreMetrics(input: CoreMetricsInput): CoreMetrics {
  const result: CoreMetrics = {
    registrationRate: missing(), attendanceRate: missing(), completionRate: missing(),
    averageViewingSeconds: missing(), postCompletionOrderRate: missing(),
    unattributedRegistrations: null, provisionalConversions: null,
  };
  try {
    validateScopeAndWindow(input);
    if (input.registrations === null) return result;
    const cohort = buildCohort(input);
    registrationMetrics(input, cohort, result);
    if (input.viewing === null) return result;
    requireFact(input.durationMs === null || (Number.isSafeInteger(input.durationMs) && input.durationMs > 0));
    const completed = viewingMetrics(input, cohort, result);
    orderMetrics(input, cohort, completed, result);
    return result;
  } catch (error) {
    if (!(error instanceof InvalidMetricFacts)) throw error;
    return { registrationRate: invalid(), attendanceRate: invalid(), completionRate: invalid(), averageViewingSeconds: invalid(), postCompletionOrderRate: invalid(), unattributedRegistrations: null, provisionalConversions: null };
  }
}


export type CompletionProjectionInput = Pick<CoreMetricsInput, "scope" | "startMs" | "endMs" | "durationMs" | "registrations" | "viewing">;
export type CompletionProjection = { status: "available"; cohortCount: number; rows: CompletionFact[] }
  | { status: "unavailable"; reason: "missing_data" | "invalid_data" | "limit_exceeded" };

/** Worker-only projection candidate. Availability is not durable provenance:
 * the writer must atomically persist rows plus manifest against the unchanged
 * source revision before any report adapter can use these completion facts.
 */
export function calculateCompletionProjection(value: CompletionProjectionInput): CompletionProjection {
  const input: CoreMetricsInput = { ...value, visits: null, completions: null, orders: null };
  try {
    validateScopeAndWindow(input);
    if (input.registrations === null || input.viewing === null || input.durationMs === null)
      return { status: "unavailable", reason: "missing_data" };
    requireFact(Number.isSafeInteger(input.durationMs) && input.durationMs > 0);
    const cohort = buildCohort(input);
    const result: CoreMetrics = { registrationRate: missing(), attendanceRate: missing(), completionRate: missing(),
      averageViewingSeconds: missing(), postCompletionOrderRate: missing(), unattributedRegistrations: null, provisionalConversions: null };
    const completed = viewingMetrics(input, cohort, result);
    if (result.completionRate.status === "unavailable") return result.completionRate;
    return { status: "available", cohortCount: cohort.size,
      rows: [...completed].sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0)

  } catch (error) {

import { COMPLETION_PERCENT } from "./live-metric-contract";

/** These are bounded exposure samples, not proof of human attention. A caller
 * must reset on seek, buffering, pause, source/version changes and rate changes.
 * Monotonic elapsed time is later anchored to server time by ingestion.
 */
export type PlaybackSample = Readonly<{
  elapsedMs: number; mediaMs: number; rate: number; playing: boolean;
}>;
export type PlaybackSegment = Readonly<{
  wallStartMs: number; wallEndMs: number; mediaStartMs: number; mediaEndMs: number;
}>;
export const MAX_PLAYBACK_SAMPLE_GAP_MS = 5000;
export const MAX_COMPLETION_SEGMENTS = 10000;

function validSample(sample: PlaybackSample): boolean {
  return !!sample && Number.isSafeInteger(sample.elapsedMs) && sample.elapsedMs >= 0
    && Number.isSafeInteger(sample.mediaMs) && sample.mediaMs >= 0
    && Number.isFinite(sample.rate) && sample.rate >= 0.25 && sample.rate <= 4
    && typeof sample.playing === "boolean";
}

/** A discontinuity returns no interval. Do not connect endpoints across gaps.
 * Integer millisecond samples allow one millisecond of quantization error.
 * This check bounds accidental seek jumps; the server still validates batches.
 */
export function playbackSegment(previous: PlaybackSample | null, current: PlaybackSample): PlaybackSegment | null {
  if (!previous || !validSample(previous) || !validSample(current) || !previous.playing || !current.playing || previous.rate !== current.rate) return null;
  const elapsed = current.elapsedMs - previous.elapsedMs;
  const media = current.mediaMs - previous.mediaMs;
  if (elapsed <= 0 || elapsed > MAX_PLAYBACK_SAMPLE_GAP_MS || media <= 0 || Math.abs(media - elapsed * current.rate) > 1) return null;
  return { wallStartMs: previous.elapsedMs, wallEndMs: current.elapsedMs, mediaStartMs: previous.mediaMs, mediaEndMs: current.mediaMs };
}

/** Derive the earliest integer millisecond at which unique coverage reached
 * 90%. Segments must be validated constant-rate exposure for one person/version.
 * Sorting by arrival order is incorrect for delayed and overlapping devices;
 * instead use a monotone coverage predicate over playback time.
 */
export function firstCompletionTime(durationMs: number | null, segments: readonly PlaybackSegment[] | null, window: Readonly<{ startMs: number; endMs: number }> = { startMs: 0, endMs: 8_640_000_000_000_000 }):
  | { status: "unavailable"; reason: "missing_data" | "invalid_data" | "limit_exceeded" }
  | { status: "available"; firstCompletedAtMs: number | null } {
  if (durationMs === null || segments === null) return { status: "unavailable", reason: "missing_data" };
  if (!Array.isArray(segments) || !Number.isSafeInteger(durationMs) || durationMs <= 0) return { status: "unavailable", reason: "invalid_data" };
  if (![window.startMs, window.endMs].every((value) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000) || window.startMs > window.endMs) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length > MAX_COMPLETION_SEGMENTS) return { status: "unavailable", reason: "limit_exceeded" };
  if (Array.from(segments).some((row) => !validSegment(row, durationMs))) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length === 0) return { status: "available", firstCompletedAtMs: null };
  let low = Infinity;
  let high = 0;
  for (const row of segments) {
    low = Math.min(low, row.wallStartMs);
    high = Math.max(high, row.wallEndMs);
  }
  low = Math.max(low, window.startMs);
  high = Math.min(high, window.endMs);
  const prepared = prepareSegments(segments, window.startMs);
  if (!prepared) return { status: "unavailable", reason: "limit_exceeded" };
  if (low > high || !completedAt(durationMs, prepared, high)) return { status: "available", firstCompletedAtMs: null };
  // At most 53 iterations for safe integer time. No per-millisecond scan and
  // no invented first-completion timestamp from receipt/arrival order.
  while (low < high) {
    const middle = low + Math.floor((high - low) / 2);
    if (completedAt(durationMs, prepared, middle)) high = middle;
    else low = middle + 1;
  }
  return { status: "available", firstCompletedAtMs: low };
}

function validSegment(row: PlaybackSegment, durationMs: number): boolean {
  return !!row && [row.wallStartMs, row.wallEndMs, row.mediaStartMs, row.mediaEndMs].every((value) => Number.isSafeInteger(value) && value >= 0)
    && row.wallEndMs <= 8_640_000_000_000_000 && row.wallEndMs > row.wallStartMs
    && row.mediaEndMs > row.mediaStartMs && row.mediaEndMs <= durationMs;
}

function gcd(a: bigint, b: bigint): bigint {
  while (b !== BigInt(0)) { const next = a % b; a = b; b = next; }
  return a;
}
type PreparedSegment = { row: PlaybackSegment; start: bigint; base: bigint; rate: bigint; eligibleAtMs: number };
type PreparedHistory = { scale: bigint; segments: PreparedSegment[] };
const MAX_RATIONAL_SCALE = BigInt(1) << BigInt(8192);

function prepareSegments(segments: readonly PlaybackSegment[], eligibleAtMs: number): PreparedHistory | null {
  const eligible = segments.filter((row) => row.wallEndMs > eligibleAtMs);
  // One common exact scale avoids huge repeated fraction reductions inside
  // binary search. Bound its size before allocating scaled per-segment values.
  let scale = BigInt(1);
  for (const row of eligible) {
    const duration = BigInt(row.wallEndMs - row.wallStartMs);
    scale = scale / gcd(scale, duration) * duration;
    if (scale > MAX_RATIONAL_SCALE) return null;
  }
  const prepared = eligible.map((row) => {
    const base = BigInt(row.mediaStartMs) * scale;
    const rate = BigInt(row.mediaEndMs - row.mediaStartMs) * (scale / BigInt(row.wallEndMs - row.wallStartMs));
    const at = Math.max(row.wallStartMs, eligibleAtMs);
    return { row, base, rate, eligibleAtMs: at, start: base + rate * BigInt(at - row.wallStartMs) };
  }).sort((a, b) => a.start < b.start ? -1 : a.start > b.start ? 1 : 0);
  return { scale, segments: prepared };
}

/** Preserve original slopes and exact coverage. The fixed denominator is
 * shared by every interval, so union and threshold tests use integers only.
 * Sort eligible starts once and stop summing once the threshold is reached.
 */
function completedAt(durationMs: number, history: PreparedHistory, atMs: number): boolean {
  let end = BigInt(0);
  let covered = BigInt(0);
  const threshold = BigInt(durationMs) * BigInt(COMPLETION_PERCENT) * history.scale;
  for (const prepared of history.segments) {
    if (prepared.eligibleAtMs >= atMs) continue;
    const intervalEnd = prepared.base + prepared.rate * BigInt(Math.min(prepared.row.wallEndMs, atMs) - prepared.row.wallStartMs);
    const start = prepared.start > end ? prepared.start : end;
    if (intervalEnd > start) {
      covered += intervalEnd - start;
      end = intervalEnd;
      if (covered * BigInt(100) >= threshold) return true;
    }
  }
  return false;
}

/** Own a fresh facade for exactly one read-only RR callback. The private cache
 * cannot be supplied by callers, reused on the original client, or survive the
 * callback. Lease expiry is still checked against the DB clock on every page.
 */
export async function withMetricOrderRevision<T>(tx: ReadClient, value: MetricReportLease,
  read: (snapshot: ReadClient) => Promise<T>): Promise<T> {
  const scope = LeaseRequest.parse(value);
  await tx.$queryRaw(Prisma.sql`SET TRANSACTION READ ONLY`);
  const [state] = await tx.$queryRaw<{ isolation: string; readOnly: string; revision: bigint }[]>(Prisma.sql`
    SELECT current_setting('transaction_isolation') AS isolation,
      current_setting('transaction_read_only') AS "readOnly",
      public.wp20_metric_order_revision(${scope.vendorId},${scope.liveId}) AS revision`);
  if (!state || state.isolation !== "repeatable read" || state.readOnly !== "on"
    || typeof state.revision !== "bigint" || state.revision < BigInt(0)) throw new Error("Metric revision requires read-only repeatable-read snapshot");
  const snapshot: ReadClient = { $queryRaw: tx.$queryRaw.bind(tx) };
  snapshotRevisions.set(snapshot, { vendorId: scope.vendorId, liveId: scope.liveId, revision: state.revision });
  try { return await read(snapshot); }
  finally { snapshotRevisions.delete(snapshot); }
}

/** Shared by all sessions in one report, not reset per person/page. These are
 * work limits, not sampling or truncation rules. The caller must also bound the
 * database transaction timeout because a budget cannot cancel an active query.
 */
export class MetricReadBudget {
  private rows = 0;
  private pages = 0;
  private readonly started: number;
  constructor(private readonly limits = { rows: 1_000_000, pages: 4000, milliseconds: 30000 }, private readonly now = () => performance.now()) {
    if (!Number.isSafeInteger(limits.rows) || limits.rows < 1 || limits.rows > 1_000_000
      || !Number.isSafeInteger(limits.pages) || limits.pages < 1 || limits.pages > 4000
      || !Number.isSafeInteger(limits.milliseconds) || limits.milliseconds < 1 || limits.milliseconds > 30000) throw new Error("Invalid metric read budget");
    this.started = now();
  }
  active() { const elapsed = this.now() - this.started; return Number.isFinite(elapsed) && elapsed >= 0 && elapsed < this.limits.milliseconds; }
  nextPage() { if (!this.active() || this.pages >= this.limits.pages) return false; this.pages++; return true; }
  retain(count: number) {
    if (!Number.isSafeInteger(count) || count < 0 || !this.active() || count > this.limits.rows - this.rows) return false;
    this.rows += count; return true;
  }
}

export async function readMetricSessionIntervals(tx: Pick<Prisma.TransactionClient, "$queryRaw">,
  value: Omit<ReadRequest, "afterSequence" | "take">, budget: MetricReadBudget) {
  const rows: MetricIntervalRow[] = [];
  let afterSequence = -1;
  while (budget.nextPage()) {
    const page = await readMetricIntervalPage(tx, { ...value, afterSequence, take: 500 });
    if (page.status !== "available") return page;
    if (!budget.retain(page.rows.length)) break;
    rows.push(...page.rows);
    if (!page.hasMore) return { status: "available" as const, rows, registrationId: page.registrationId, deliveryComplete: page.deliveryComplete };
    afterSequence = page.nextSequence;
  }
  // Do not leak a prefix as if it represented all viewing for this session.
  return { status: "unavailable" as const, reason: "limit_exceeded" as const };
}

/** Read one indexed session/sequence page inside the worker's repeatable-read
 * snapshot. Never use OFFSET or an unbounded cohort query. Claim/finish happen
 * in separate read-committed transactions, as required by the write guards.

export async function readMetricIntervalPage(tx: Pick<Prisma.TransactionClient, "$queryRaw">, value: ReadRequest) {
  const input = Request.parse(value);
  const [session] = await tx.$queryRaw<SessionState[]>(Prisma.sql`
    SELECT current_setting('transaction_isolation') AS isolation, s."registrationId",s."lastSequence",s.complete,s."closedAt"
    FROM "LiveMetricSession" s
    JOIN "LiveMetricReport" r ON r."vendorId"=s."vendorId" AND r."liveId"=s."liveId" AND r."versionId"=s."versionId"

    WHERE s.id=${input.sessionId} AND s."vendorId"=${input.vendorId} AND s."liveId"=${input.liveId} AND s."versionId"=${input.versionId}
      AND r.id=${input.reportId} AND r.status='running' AND r."leaseOwner"=${input.workerId}
      AND r."leaseExpiresAt">clock_timestamp() AND r."expiresAt">clock_timestamp()
      AND r."inputRevision"=v."sourceRevision"
      AND r."inputOrderRevision"=${orderRevision(tx, input)}`);
  if (!session) return { status: "unavailable" as const, reason: "lease_or_scope_unavailable" as const };
  if (session.isolation !== "repeatable read") throw new Error("Metric reader requires repeatable-read snapshot");
  if (input.afterSequence > session.lastSequence) throw new Error("Metric cursor exceeds snapshot");
  const rows = await tx.$queryRaw<MetricIntervalRow[]>(Prisma.sql`
    SELECT sequence,"wallStartAt","wallEndAt","mediaStartMs","mediaEndMs" FROM "LiveMetricInterval"

      AND sequence>${input.afterSequence} ORDER BY sequence ASC LIMIT ${input.take + 1}`);
  if (rows.length > input.take + 1 || rows.some((row, index) => row.sequence !== input.afterSequence + index + 1)) throw new Error("Metric interval snapshot is inconsistent");
  const hasMore = rows.length > input.take;
  const page = rows.slice(0, input.take);
  const nextSequence = page.at(-1)?.sequence ?? input.afterSequence;
  if (!hasMore && nextSequence !== session.lastSequence) throw new Error("Metric interval snapshot is incomplete");
  return { status: "available" as const, registrationId: session.registrationId, rows: page, nextSequence, hasMore,
    // Complete pagination and complete delivery are different facts.
    deliveryComplete: session.complete && session.closedAt !== null };

    await expect(readMetricCompletion(db, lease)).rejects.toThrow("repeatable-read");
    await expect(readMetricCompletion(db, lease)).rejects.toThrow();
  });
  it("does not fetch the payload when the shared row budget is exhausted", async () => {
    const db = { $queryRaw: vi.fn().mockResolvedValue([{ isolation: "repeatable read", rowCount: 10000, cohortCount: 10000 }]) };
    const reserve = vi.fn().mockReturnValue(false);
    expect(await readMetricCompletion(db, lease, reserve)).toEqual({ status: "unavailable", reason: "limit_exceeded" });
    expect(reserve).toHaveBeenCalledWith(10001);
    expect(db.$queryRaw).toHaveBeenCalledTimes(1);
    expect(db.$queryRaw.mock.calls[0][0].sql).not.toContain("jsonb_array_length");
    expect(db.$queryRaw.mock.calls[0][0].sql).not.toContain("SELECT rows");
  });
  it("rejects foreign candidates before SQL and retains DB lease rejection", async () => {
    const query = vi.fn().mockResolvedValue([{ changed: 0 }]);
    const db = { $transaction: vi.fn(async (run: (tx: { $queryRaw: typeof query }) => Promise<boolean>) => run({ $queryRaw: query })) };
    await expect(publishMetricCompletion(db as unknown as Parameters<typeof publishMetricCompletion>[0], lease, BigInt(1), scope, { status: "available", cohortCount: 1, rows: [{ scope: { ...scope, vendorId: "foreign" }, personId: "p", firstCompletedAtMs: 10 }] })).rejects.toThrow("scope");
    expect(db.$transaction).not.toHaveBeenCalled();
    expect(await publishMetricCompletion(db as unknown as Parameters<typeof publishMetricCompletion>[0], lease, BigInt(1), scope, { status: "available", cohortCount: 1, rows: [] })).toBe(false);
  });
});
