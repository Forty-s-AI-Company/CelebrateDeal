200: /** Shared by all sessions in one report, not reset per person/page. These are
201:  * work limits, not sampling or truncation rules. The caller must also bound the
202:  * database transaction timeout because a budget cannot cancel an active query.

204: export class MetricReadBudget {
205:   private rows = 0;
206:   private pages = 0;
207:   private readonly started: number;
208:   constructor(private readonly limits = { rows: 1_000_000, pages: 4000, milliseconds: 30000 }, private readonly now = () => performance.now()) {
209:     if (!Number.isSafeInteger(limits.rows) || limits.rows < 1 || limits.rows > 1_000_000
210:       || !Number.isSafeInteger(limits.pages) || limits.pages < 1 || limits.pages > 4000
211:       || !Number.isSafeInteger(limits.milliseconds) || limits.milliseconds < 1 || limits.milliseconds > 30000) throw new Error("Invalid metric read budget");
212:     this.started = now();
213:   }
214:   active() { const elapsed = this.now() - this.started; return Number.isFinite(elapsed) && elapsed >= 0 && elapsed < this.limits.milliseconds; }
215:   nextPage() { if (!this.active() || this.pages >= this.limits.pages) return false; this.pages++; return true; }
216:   retain(count: number) {
217:     if (!Number.isSafeInteger(count) || count < 0 || !this.active() || count > this.limits.rows - this.rows) return false;
218:     this.rows += count; return true;
219:   }

280: -- Ordinary evidence does not rewrite a shared version counter. It invalidates
281: -- only reports whose input window overlaps the changed facts. The shared lock
282: -- serializes report creation against inserts without serializing viewers.
283: CREATE FUNCTION public.wp20_metric_evidence_changed() RETURNS trigger
284: LANGUAGE plpgsql SET search_path = pg_catalog, public SET timezone = 'UTC' AS $$
285: DECLARE fact JSONB; affected_start TIMESTAMPTZ; affected_end TIMESTAMPTZ;
286: BEGIN
287:   IF current_setting('transaction_isolation')<>'read committed' THEN
288:     RAISE EXCEPTION 'metric writes require read committed isolation' USING ERRCODE='23514';
289:   END IF;
290:   IF TG_TABLE_NAME='LiveMetricSession' AND TG_OP='UPDATE' THEN
291:     IF (to_jsonb(NEW)-'lastSequence'-'lastElapsedMs')=(to_jsonb(OLD)-'lastSequence'-'lastElapsedMs') THEN RETURN NULL; END IF;
292:   END IF;
293:   fact := CASE WHEN TG_OP='DELETE' THEN to_jsonb(OLD) ELSE to_jsonb(NEW) END;
294:   IF TG_TABLE_NAME='LiveMetricInterval' THEN
295:     affected_start := (fact->>'wallStartAt')::timestamp;
296:     affected_end := (fact->>'wallEndAt')::timestamp;
297:   ELSIF TG_TABLE_NAME='LiveMetricVisitor' THEN
298:     affected_start := (fact->>'firstVisitedAt')::timestamp;
299:     affected_end := affected_start+INTERVAL '1 millisecond';
300:   ELSIF TG_TABLE_NAME='LiveMetricRegistration' THEN
301:     affected_start := (fact->>'verifiedAt')::timestamp;
302:     affected_end := affected_start+INTERVAL '1 millisecond';
303:   ELSE
304:     affected_start := (fact->>'anchorAt')::timestamp;
305:     affected_end := (fact->>'expiresAt')::timestamp;
306:   END IF;
307:   PERFORM 1 FROM public."LiveMetricVersion" WHERE id=fact->>'versionId'
308:     AND "vendorId"=fact->>'vendorId' AND "liveId"=fact->>'liveId' FOR SHARE;
309:   IF NOT FOUND THEN RETURN NULL; END IF;
310:   UPDATE public."LiveMetricReport" SET status='stale',result=NULL,"leaseOwner"=NULL,"leaseExpiresAt"=NULL
311:     WHERE "versionId"=fact->>'versionId' AND "vendorId"=fact->>'vendorId' AND "liveId"=fact->>'liveId'
312:       AND "windowStartAt"<affected_end AND "windowEndAt">affected_start AND status<>'stale';
313:   RETURN NULL;
314: END $$;
315: DO $$ DECLARE metric_table TEXT; BEGIN
316:   FOREACH metric_table IN ARRAY ARRAY['LiveMetricVisitor','LiveMetricRegistration','LiveMetricSession','LiveMetricInterval'] LOOP
317:     EXECUTE format('CREATE TRIGGER %I BEFORE UPDATE ON public.%I FOR EACH ROW EXECUTE FUNCTION public.wp20_metric_scope_update()', metric_table || '_scope_immutable', metric_table);
318:     EXECUTE format('CREATE TRIGGER %I AFTER INSERT OR UPDATE OR DELETE ON public.%I FOR EACH ROW EXECUTE FUNCTION public.wp20_metric_evidence_changed()', metric_table || '_revision', metric_table);
319:   END LOOP;
320: END $$;

351: CREATE FUNCTION public.wp20_metric_interval_guard() RETURNS trigger
352: LANGUAGE plpgsql SET search_path = pg_catalog, public SET timezone = 'UTC' AS $$
353: DECLARE current_session public."LiveMetricSession"%ROWTYPE;
354: BEGIN
355:   IF current_setting('transaction_isolation')<>'read committed' THEN
356:     RAISE EXCEPTION 'metric writes require read committed isolation' USING ERRCODE='23514';
357:   END IF;
358:   IF TG_OP='DELETE' THEN
359:     -- Parent cascade is allowed; deleting individual evidence would punch a
360:     -- hole in a session that has already proved its contiguous completion.
361:     IF EXISTS (SELECT 1 FROM public."LiveMetricSession" WHERE id=OLD."sessionId") THEN
362:       RAISE EXCEPTION 'metric interval deletion requires parent cascade' USING ERRCODE='23514';
363:     END IF;
364:     RETURN OLD;
365:   END IF;
366:   IF TG_OP = 'UPDATE' THEN
367:     RAISE EXCEPTION 'metric interval is append only' USING ERRCODE = '23514';
368:   END IF;
369:   -- Lock mutable authorization sources before the metric session. Revocation,
370:   -- release and source changes either precede this insert or wait for it.
371:   SELECT * INTO current_session FROM public."LiveMetricSession" WHERE "id"=NEW."sessionId";
372:   IF NOT FOUND THEN RAISE EXCEPTION 'metric session missing' USING ERRCODE='23514'; END IF;
373:   PERFORM 1 FROM public."Live" WHERE id=NEW."liveId" AND "vendorId"=NEW."vendorId" FOR SHARE;
374:   PERFORM 1 FROM public."FormSubmission" f JOIN public."LiveMetricRegistration" r ON r."submissionId"=f.id
375:     WHERE r.id=current_session."registrationId" FOR SHARE OF f;
376:   PERFORM 1 FROM public."LiveViewerSession" WHERE id=current_session."admissionSessionId" FOR SHARE;
377:   PERFORM 1 FROM public."LiveMetricVersion" WHERE id=NEW."versionId"
378:     AND "vendorId"=NEW."vendorId" AND "liveId"=NEW."liveId" FOR SHARE;
379:   SELECT * INTO current_session FROM public."LiveMetricSession" WHERE "id"=NEW."sessionId" FOR UPDATE;
380:   -- Check expiry after any lock wait, against the database-owned receipt time.
381:   NEW."receivedAt" := clock_timestamp();
382:   IF NOT FOUND OR NEW."sequence"<>current_session."lastSequence"+1
383:     OR (current_session."finalSequence" IS NOT NULL AND NEW."sequence">current_session."finalSequence")
384:     OR NEW."wallStartAt"<current_session."anchorAt" + current_session."lastElapsedMs" * INTERVAL '1 millisecond' THEN
385:     RAISE EXCEPTION 'metric intervals must be contiguous and nonoverlapping' USING ERRCODE='23514';
386:   END IF;
387:   IF NOT EXISTS (
388:     SELECT 1 FROM public."LiveMetricSession" s
389:     JOIN public."LiveMetricVersion" v ON v."id" = s."versionId" AND v."vendorId" = s."vendorId" AND v."liveId" = s."liveId"
390:     JOIN public."Live" l ON l."id"=v."liveId" AND l."vendorId"=v."vendorId" AND l."videoId" IS NOT DISTINCT FROM v."videoId"
391:     JOIN public."LiveMetricRegistration" r ON r."id" = s."registrationId" AND r."vendorId" = s."vendorId" AND r."liveId" = s."liveId" AND r."versionId" = s."versionId"
392:     JOIN public."FormSubmission" f ON f."id" = r."submissionId" AND f."liveId" = r."liveId"
393:     JOIN public."LiveViewerSession" a ON a."id" = s."admissionSessionId" AND a."vendorId" = s."vendorId" AND a."liveId" = s."liveId"
394:     WHERE s."id" = NEW."sessionId" AND s."vendorId" = NEW."vendorId" AND s."liveId" = NEW."liveId" AND s."versionId" = NEW."versionId"
395:       AND v."collectionEnabled" AND NEW."wallStartAt" >= v."collectionStartedAt" AND NEW."receivedAt" < v."expiresAt"
396:       AND s."closedAt" IS NULL AND NOT s."complete" AND NEW."wallStartAt" >= s."anchorAt"
397:       AND NEW."wallStartAt" >= r."verifiedAt" AND NEW."receivedAt" < s."expiresAt" AND NEW."receivedAt" < a."expiresAt"
398:       AND f."verificationStatus" = 'VERIFIED' AND f."verifiedAt" = r."verifiedAt" AND l."formId"=f."formId"
399:       AND (v."durationMs" IS NULL OR NEW."mediaEndMs" <= v."durationMs")
400:   ) THEN
401:     RAISE EXCEPTION 'metric interval scope or lifetime invalid' USING ERRCODE = '23514';
402:   END IF;
403:   RETURN NEW;
404: END $$;

518: LANGUAGE plpgsql SET search_path = pg_catalog, public SET timezone = 'UTC' AS $$
519: DECLARE current_revision BIGINT;
520: BEGIN
521:   IF current_setting('transaction_isolation')<>'read committed' THEN
522:     RAISE EXCEPTION 'metric writes require read committed isolation' USING ERRCODE='23514';

563: -- Worker entry points lock the version before the report, matching ingestion's
564: -- version -> report invalidation order. The API must not claim with raw UPDATE.
565: CREATE FUNCTION public.wp20_metric_report_claim(
566:   vendor_id TEXT, live_id TEXT, version_id TEXT, report_id TEXT,
567:   worker_id TEXT, lease_until TIMESTAMPTZ(3)
568: ) RETURNS INTEGER LANGUAGE plpgsql SET search_path = pg_catalog, public SET timezone = 'UTC' AS $$
569: DECLARE changed INTEGER; previous_writer TEXT;
570: BEGIN
571:   IF worker_id IS NULL OR length(worker_id)=0 OR lease_until<=clock_timestamp()
572:     OR lease_until>clock_timestamp()+INTERVAL '5 minutes' THEN
573:     RAISE EXCEPTION 'invalid metric worker lease' USING ERRCODE='23514';
574:   END IF;
575:   PERFORM 1 FROM public."LiveMetricVersion"
576:     WHERE id=version_id AND "vendorId"=vendor_id AND "liveId"=live_id FOR UPDATE;
577:   IF NOT FOUND THEN RETURN 0; END IF;
578:   UPDATE public."LiveMetricReport" SET status='failed',"errorCode"='attempt_limit',"leaseOwner"=NULL,"leaseExpiresAt"=NULL
579:     WHERE id=report_id AND "vendorId"=vendor_id AND "liveId"=live_id AND "versionId"=version_id
580:       AND attempts>=5 AND (status='queued' OR (status='running' AND "leaseExpiresAt"<=clock_timestamp()));
581:   previous_writer := current_setting('wp20.report_writer',true);
582:   PERFORM set_config('wp20.report_writer','claim',true);
583:   UPDATE public."LiveMetricReport" SET status='running',"leaseOwner"=worker_id,
584:     "leaseExpiresAt"=lease_until,attempts=attempts+1
585:     WHERE id=report_id AND "vendorId"=vendor_id AND "liveId"=live_id AND "versionId"=version_id
586:       AND (status='queued' OR (status='running' AND "leaseExpiresAt"<=clock_timestamp()))
587:       AND attempts<5 AND "expiresAt">clock_timestamp();
588:   GET DIAGNOSTICS changed = ROW_COUNT;
589:   PERFORM set_config('wp20.report_writer',coalesce(previous_writer,''),true);
590:   RETURN changed;

155: CREATE OR REPLACE FUNCTION public.wp20_metric_completion_publish(
156:  vendor_id TEXT,live_id TEXT,version_id TEXT,report_id TEXT,worker_id TEXT,
157:  source_revision BIGINT,cohort_count INTEGER,facts JSONB
158: ) RETURNS INTEGER LANGUAGE plpgsql SECURITY DEFINER
159: SET search_path=pg_catalog,public SET timezone='UTC' AS $$
160: DECLARE report public."LiveMetricReport"%ROWTYPE; current_revision BIGINT; actual_cohort BIGINT;
161: BEGIN
162:  IF current_setting('transaction_isolation')<>'read committed' THEN
163:   RAISE EXCEPTION 'completion write requires read committed' USING ERRCODE='23514';
164:  END IF;
165:  IF facts IS NULL OR jsonb_typeof(facts)<>'array' OR cohort_count IS NULL OR cohort_count<0
166:    OR source_revision IS NULL OR source_revision<0 THEN
167:   RAISE EXCEPTION 'invalid completion manifest' USING ERRCODE='23514';
168:  END IF;
169:  IF jsonb_array_length(facts)>cohort_count OR cohort_count>1000000 OR pg_column_size(facts)>67108864 THEN
170:   RAISE EXCEPTION 'completion manifest limit exceeded' USING ERRCODE='23514';
171:  END IF;
172:  -- Input-only checks require no locks and reject malformed batches early.
173:  IF EXISTS(SELECT 1 FROM jsonb_array_elements(facts) f WHERE jsonb_typeof(f)<>'object') THEN
174:   RAISE EXCEPTION 'invalid completion fact shape' USING ERRCODE='23514';
175:  END IF;
176:  IF EXISTS(SELECT 1 FROM jsonb_array_elements(facts) f WHERE
177:    NOT (f ? 'personId' AND f ? 'firstCompletedAtMs')
178:    OR jsonb_typeof(f->'personId')<>'string' OR jsonb_typeof(f->'firstCompletedAtMs')<>'number'
179:    OR (SELECT count(*) FROM jsonb_object_keys(f))<>2) THEN
180:   RAISE EXCEPTION 'invalid completion fact shape' USING ERRCODE='23514';
181:  END IF;
182:  IF (SELECT count(DISTINCT f->>'personId') FROM jsonb_array_elements(facts) f)<>jsonb_array_length(facts) THEN
183:   RAISE EXCEPTION 'duplicate completion person' USING ERRCODE='23514';
184:  END IF;
185:  IF EXISTS(SELECT 1 FROM jsonb_array_elements(facts) f WHERE
186:   (f->>'firstCompletedAtMs')::numeric<>trunc((f->>'firstCompletedAtMs')::numeric)
187:   OR (f->>'firstCompletedAtMs')::numeric<0 OR (f->>'firstCompletedAtMs')::numeric>8640000000000000) THEN
188:   RAISE EXCEPTION 'completion fact outside timestamp range' USING ERRCODE='23514';
189:  END IF;
190:  -- Match the established version -> report lock order before publishing.
191:  SELECT "sourceRevision" INTO current_revision FROM public."LiveMetricVersion"
192:   WHERE id=version_id AND "vendorId"=vendor_id AND "liveId"=live_id FOR UPDATE;
193:  IF NOT FOUND OR current_revision<>source_revision THEN RETURN 0; END IF;
194:  SELECT * INTO report FROM public."LiveMetricReport" WHERE id=report_id
195:   AND "vendorId"=vendor_id AND "liveId"=live_id AND "versionId"=version_id FOR UPDATE;
196:  IF NOT FOUND OR report.status<>'running' OR report."leaseOwner" IS DISTINCT FROM worker_id
197:   OR report."leaseExpiresAt"<=clock_timestamp() OR report."expiresAt"<=clock_timestamp()
198:   OR report."inputRevision"<>source_revision
199:   OR report."inputOrderRevision" IS DISTINCT FROM public.wp20_metric_order_revision(vendor_id,live_id)
200:   OR report."inputRegistrationRevision" IS DISTINCT FROM public.wp20_metric_registration_revision(vendor_id,live_id)
201:  THEN RETURN 0; END IF;
202:  SELECT count(*) INTO actual_cohort FROM public."LiveMetricRegistration"
203:   WHERE "vendorId"=vendor_id AND "liveId"=live_id AND "versionId"=version_id
204:     AND "verifiedAt">=report."windowStartAt" AND "verifiedAt"<report."windowEndAt";
205:  IF actual_cohort<>cohort_count THEN RAISE EXCEPTION 'completion cohort changed' USING ERRCODE='23514'; END IF;
206:  -- Scope-dependent membership remains under the version/report locks.
207:  IF EXISTS(
208:   SELECT 1 FROM jsonb_to_recordset(facts) AS f("personId" text,"firstCompletedAtMs" numeric)
209:   LEFT JOIN public."LiveMetricRegistration" g
210:    ON g.id=f."personId" AND g."vendorId"=vendor_id AND g."liveId"=live_id AND g."versionId"=version_id
211:     AND g."verifiedAt">=report."windowStartAt" AND g."verifiedAt"<report."windowEndAt"
212:     AND f."firstCompletedAtMs">=extract(epoch FROM g."verifiedAt")*1000
213:     AND f."firstCompletedAtMs"<extract(epoch FROM report."windowEndAt")*1000
214:   WHERE g.id IS NULL) THEN
215:   RAISE EXCEPTION 'completion fact outside cohort window' USING ERRCODE='23514';
216:  END IF;
217:  -- Validation may consume the rest of a short lease; do not publish after expiry.
218:  IF report."leaseExpiresAt"<=clock_timestamp() OR report."expiresAt"<=clock_timestamp() THEN RETURN 0; END IF;
219:  INSERT INTO public."LiveMetricCompletionProjection" ("reportId","sourceRevision","algorithmVersion","cohortCount","rowCount",rows)
220:  VALUES (report_id,source_revision,1,cohort_count,jsonb_array_length(facts),facts)
221:  ON CONFLICT ("reportId") DO NOTHING;
222:  IF NOT EXISTS(SELECT 1 FROM public."LiveMetricCompletionProjection" p WHERE p."reportId"=report_id
223:   AND p."sourceRevision"=source_revision AND p."cohortCount"=cohort_count AND p.rows=facts) THEN
224:   RAISE EXCEPTION 'completion projection conflict' USING ERRCODE='23514';
225:  END IF;
226:  RETURN 1;
227: END $$;
