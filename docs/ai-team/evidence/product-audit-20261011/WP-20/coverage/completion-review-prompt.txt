/** Internal writer: caller must establish complete registration/viewing coverage
 * and delivery before calculating candidates. The DB rechecks lease/revision;
 * a worker result alone is not permission to certify missing source history.
 */
export async function publishMetricCompletion(db: Database, value: MetricReportLease, sourceRevision: bigint,
  scope: MetricScope, projection: CompletionProjection): Promise<boolean> {
  const lease = leaseSchema.parse(value);
  if (typeof sourceRevision !== "bigint" || sourceRevision < BigInt(0)) throw new Error("Invalid completion source revision");
  if (scope.vendorId !== lease.vendorId || scope.liveId !== lease.liveId || scope.contentVersionId !== lease.versionId
    || !["live", "replay"].includes(scope.mode)) throw new Error("Completion scope mismatch");
  if (projection.status !== "available") return false;
  const cohortCount = z.number().int().min(0).max(1_000_000).parse(projection.cohortCount);
  if (projection.rows.some(row => row.scope.vendorId !== scope.vendorId || row.scope.liveId !== scope.liveId
    || row.scope.contentVersionId !== scope.contentVersionId || row.scope.mode !== scope.mode)) throw new Error("Completion scope mismatch");
  const rows = factsSchema.parse(projection.rows.map(row => ({ personId: row.personId, firstCompletedAtMs: row.firstCompletedAtMs })));
  if (rows.length > cohortCount) throw new Error("Completion cohort mismatch");
  const [result] = await db.$queryRaw<{ changed: number }[]>(Prisma.sql`
    SELECT public.wp20_metric_completion_publish(${lease.vendorId},${lease.liveId},${lease.versionId},${lease.reportId},${lease.workerId},
      ${sourceRevision}::bigint,${cohortCount}::integer,${JSON.stringify(rows)}::jsonb) AS changed`);
  return result?.changed === 1;
}

type Database = Pick<PrismaClient, "$queryRaw" | "$transaction">;
type Executor = ReturnType<typeof createMetricExecutor>;

/** Internal single-job orchestration; caller authenticates and chooses a scoped
 * queued report. Supply the bounded worker executor, never a browser callback.
 * No automatic retries or synchronous computation fallback. Unfinished leases
 * expire through the existing DB-clock protocol so later attempts can recover.
 */
export async function runMetricReportJob(db: Database, value: MetricReportLease, execute: Executor, project?: ReturnType<typeof createCompletionExecutor>) {
  const lease = { ...value };
  let stage: "claim" | "snapshot" | "conversion" | "execution" | "projection" | "publish" | "reload" | "finish" = "claim";
  try {
    if (!await claimMetricReport(db, lease)) return { status: "not_claimed" as const };
    stage = "snapshot";
    let snapshot = await loadMetricViewingSnapshot(db, lease);
    if (snapshot.status !== "available") return { status: "snapshot_unavailable" as const, reason: snapshot.reason };
    stage = "conversion";
    let input = metricSnapshotInput(snapshot.data);
    if (project && input.completions === null && input.registrations !== null && input.viewing !== null && input.durationMs !== null) {
      const sourceRevision = snapshot.data.context.sourceRevision;
      if (typeof sourceRevision !== "bigint") throw new Error("Missing snapshot revision");
      stage = "projection";
      const candidate = await project(input);
      if (candidate.status !== "available") return { status: "execution_unavailable" as const, reason: candidate.reason };
      if (candidate.projection.status === "available") {
        stage = "publish";
        if (!await publishMetricCompletion(db, lease, sourceRevision, input.scope, candidate.projection)) return { status: "lease_lost" as const };
        stage = "reload";
        snapshot = await loadMetricViewingSnapshot(db, lease);
        if (snapshot.status !== "available") return { status: "snapshot_unavailable" as const, reason: snapshot.reason };
        input = metricSnapshotInput(snapshot.data);
        if (input.completions === null || snapshot.data.context.sourceRevision !== sourceRevision) return { status: "lease_lost" as const };
      }
    }
    stage = "execution";
    const result = await execute(input);
    if (result.status !== "available") return { status: "execution_unavailable" as const, reason: result.reason };
    stage = "finish";
    // Finish revalidates the lease and source revision after expensive work.
    // Only aggregate output is persisted; raw person/browser facts stay private.
    if (!await finishMetricReport(db, lease, result.metrics)) return { status: "lease_lost" as const };
    return { status: "recorded" as const };
  } catch {
    // Failure summaries carry no database errors, identities or worker payloads.
    return { status: "failed" as const, stage };
  }
}

async function appendRegistrationSessions(tx: Pick<Prisma.TransactionClient, "$queryRaw">, lease: MetricReportLease, registrationId: string,
  budget: MetricReadBudget, result: MetricViewingSnapshot) {

    if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
    if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");

}

import type { CoreMetricsInput } from "./live-core-metrics-read-model";
import type { MetricViewingSnapshot } from "./live-metric-report-snapshot";

/** Pure conversion for the isolated calculator worker. Scope comes from the
 * leased snapshot, never from a second caller-selected tenant. Coverage and
 * delivery are separate requirements; neither can stand in for the other.
 */
export function metricSnapshotInput(snapshot: MetricViewingSnapshot): CoreMetricsInput {
  const { scope, context } = snapshot;
  const registrationsComplete = snapshot.registrationCoverage.status === "complete";
  const viewingComplete = registrationsComplete && snapshot.viewingCoverage.status === "complete" && snapshot.deliveryComplete;
  if (snapshot.registrations.some(row => !row.identityCurrent)) throw new Error("Metric snapshot identity unavailable");
  const people = new Set(snapshot.registrations.map(row => row.id));
  if (snapshot.sessions.some(row => !people.has(row.registrationId))) throw new Error("Metric snapshot identity unavailable");
  const projection = snapshot.completionProjection ?? null;
  if (projection && (projection.cohortCount !== people.size || projection.rows.some(row => !people.has(row.personId))))
    throw new Error("Metric completion cohort unavailable");
  return {
    scope, startMs: context.windowStartAt.getTime(), endMs: context.windowEndAt.getTime(), durationMs: context.durationMs,
    visits: snapshot.visitCoverage.status === "complete" ? snapshot.visits.map(row => ({ scope, browserId: row.visitorId, visitedAtMs: row.visitedAt.getTime() })) : null,
    registrations: registrationsComplete ? snapshot.registrations.map(row => ({ scope, personId: row.id,
      browserId: row.visitorId, verifiedAtMs: row.verifiedAt.getTime() })) : null,
    viewing: viewingComplete ? snapshot.sessions.flatMap(row => row.intervals.map(interval => ({ scope,
      personId: row.registrationId, wallStartMs: interval.wallStartAt.getTime(), wallEndMs: interval.wallEndAt.getTime(),
      mediaStartMs: interval.mediaStartMs, mediaEndMs: interval.mediaEndMs }))) : null,
    // Completion requires durable first-crossing provenance from the SAME
    // snapshot. A certificate alone cannot turn absence into [].
    completions: viewingComplete && projection ? projection.rows.map(row => ({ scope, ...row })) : null,
    orders: snapshot.orderCoverage.status === "complete" ? snapshot.orders.map(row => ({ scope, personId: row.registrationId,
      establishedAtMs: row.establishedAt.getTime(), cancelled: row.cancelled })) : null,
  };
}

type BoundedResult<T> = { status: "available"; value: T }
  | { status: "unavailable"; reason: "busy" | "input_limit" | "timeout" | "worker_failed" };
function createBoundedExecutor<T>(artifact: URL, operation: "metrics" | "completion", timeoutMs: number) {
  if (artifact.protocol !== "file:" || !Number.isInteger(timeoutMs) || timeoutMs < 1 || timeoutMs > 30000) throw new Error("Invalid metric executor configuration");
  let active = false;
  return async (input: CoreMetricsInput): Promise<BoundedResult<T>> => {
    if (active) return { status: "unavailable", reason: "busy" };
    const lists = [input.visits, input.registrations, input.viewing, input.completions, input.orders];
    if (lists.some(rows => rows !== null && !Array.isArray(rows))
      || lists.reduce((sum, rows) => sum + (rows?.length ?? 0), 0) > 1_000_000) return { status: "unavailable", reason: "input_limit" };
    active = true;
    try {
      return await new Promise<BoundedResult<T>>(resolve => {
        let worker: Worker;
        try {
          worker = new Worker(artifact, { workerData: { operation, input }, env: {}, execArgv: [], stdout: true, stderr: true,
            resourceLimits: { maxOldGenerationSizeMb: 256, maxYoungGenerationSizeMb: 32, stackSizeMb: 4 } });
        } catch { resolve({ status: "unavailable", reason: "worker_failed" }); return; }
        // Consume and discard worker diagnostics; raw facts never enter logs.
        worker.stdout?.resume(); worker.stderr?.resume();
        let settled = false;
        const finish = (result: BoundedResult<T>) => {
          if (settled) return;
          settled = true; clearTimeout(timer);
          void worker.terminate().catch(() => undefined).then(() => resolve(result));
        };
        const timer = setTimeout(() => finish({ status: "unavailable", reason: "timeout" }), timeoutMs);
        worker.once("message", (value: T) => finish({ status: "available", value }));
        worker.once("error", () => finish({ status: "unavailable", reason: "worker_failed" }));
        worker.once("exit", () => finish({ status: "unavailable", reason: "worker_failed" }));
      });
    } finally { active = false; }
  };
}

import { COMPLETION_PERCENT } from "./live-metric-contract";

/** These are bounded exposure samples, not proof of human attention. A caller
 * must reset on seek, buffering, pause, source/version changes and rate changes.
 * Monotonic elapsed time is later anchored to server time by ingestion.
 */
export type PlaybackSample = Readonly<{
  elapsedMs: number; mediaMs: number; rate: number; playing: boolean;
}>;
export type PlaybackSegment = Readonly<{
  wallStartMs: number; wallEndMs: number; mediaStartMs: number; mediaEndMs: number;
}>;
export const MAX_PLAYBACK_SAMPLE_GAP_MS = 5000;
export const MAX_COMPLETION_SEGMENTS = 10000;

function validSample(sample: PlaybackSample): boolean {
  return !!sample && Number.isSafeInteger(sample.elapsedMs) && sample.elapsedMs >= 0
    && Number.isSafeInteger(sample.mediaMs) && sample.mediaMs >= 0
    && Number.isFinite(sample.rate) && sample.rate >= 0.25 && sample.rate <= 4
    && typeof sample.playing === "boolean";
}

/** A discontinuity returns no interval. Do not connect endpoints across gaps.
 * Integer millisecond samples allow one millisecond of quantization error.
 * This check bounds accidental seek jumps; the server still validates batches.
 */
export function playbackSegment(previous: PlaybackSample | null, current: PlaybackSample): PlaybackSegment | null {
  if (!previous || !validSample(previous) || !validSample(current) || !previous.playing || !current.playing || previous.rate !== current.rate) return null;
  const elapsed = current.elapsedMs - previous.elapsedMs;
  const media = current.mediaMs - previous.mediaMs;
  if (elapsed <= 0 || elapsed > MAX_PLAYBACK_SAMPLE_GAP_MS || media <= 0 || Math.abs(media - elapsed * current.rate) > 1) return null;
  return { wallStartMs: previous.elapsedMs, wallEndMs: current.elapsedMs, mediaStartMs: previous.mediaMs, mediaEndMs: current.mediaMs };
}

/** Derive the earliest integer millisecond at which unique coverage reached
 * 90%. Segments must be validated constant-rate exposure for one person/version.
 * Sorting by arrival order is incorrect for delayed and overlapping devices;
 * instead use a monotone coverage predicate over playback time.
 */
export function firstCompletionTime(durationMs: number | null, segments: readonly PlaybackSegment[] | null, window: Readonly<{ startMs: number; endMs: number }> = { startMs: 0, endMs: 8_640_000_000_000_000 }):
  | { status: "unavailable"; reason: "missing_data" | "invalid_data" | "limit_exceeded" }
  | { status: "available"; firstCompletedAtMs: number | null } {
  if (durationMs === null || segments === null) return { status: "unavailable", reason: "missing_data" };
  if (!Array.isArray(segments) || !Number.isSafeInteger(durationMs) || durationMs <= 0) return { status: "unavailable", reason: "invalid_data" };
  if (![window.startMs, window.endMs].every((value) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000) || window.startMs > window.endMs) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length > MAX_COMPLETION_SEGMENTS) return { status: "unavailable", reason: "limit_exceeded" };
  if (Array.from(segments).some((row) => !validSegment(row, durationMs))) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length === 0) return { status: "available", firstCompletedAtMs: null };
  let low = Infinity;
  let high = 0;
  for (const row of segments) {
    low = Math.min(low, row.wallStartMs);
    high = Math.max(high, row.wallEndMs);
  }
  low = Math.max(low, window.startMs);
  high = Math.min(high, window.endMs);
  const prepared = prepareSegments(segments, window.startMs);
  if (!prepared) return { status: "unavailable", reason: "limit_exceeded" };
  if (low > high || !completedAt(durationMs, prepared, high)) return { status: "available", firstCompletedAtMs: null };
  // At most 53 iterations for safe integer time. No per-millisecond scan and
  // no invented first-completion timestamp from receipt/arrival order.
  while (low < high) {
    const middle = low + Math.floor((high - low) / 2);
    if (completedAt(durationMs, prepared, middle)) high = middle;
    else low = middle + 1;
  }
  return { status: "available", firstCompletedAtMs: low };
}

function validSegment(row: PlaybackSegment, durationMs: number): boolean {
  return !!row && [row.wallStartMs, row.wallEndMs, row.mediaStartMs, row.mediaEndMs].every((value) => Number.isSafeInteger(value) && value >= 0)
    && row.wallEndMs <= 8_640_000_000_000_000 && row.wallEndMs > row.wallStartMs
    && row.mediaEndMs > row.mediaStartMs && row.mediaEndMs <= durationMs;
}

function gcd(a: bigint, b: bigint): bigint {
  while (b !== BigInt(0)) { const next = a % b; a = b; b = next; }
  return a;
}
type PreparedSegment = { row: PlaybackSegment; start: bigint; base: bigint; rate: bigint; eligibleAtMs: number };
type PreparedHistory = { scale: bigint; segments: PreparedSegment[] };
const MAX_RATIONAL_SCALE = BigInt(1) << BigInt(8192);

function prepareSegments(segments: readonly PlaybackSegment[], eligibleAtMs: number): PreparedHistory | null {
  const eligible = segments.filter((row) => row.wallEndMs > eligibleAtMs);
  // One common exact scale avoids huge repeated fraction reductions inside
  // binary search. Bound its size before allocating scaled per-segment values.
  let scale = BigInt(1);
  for (const row of eligible) {
    const duration = BigInt(row.wallEndMs - row.wallStartMs);
    scale = scale / gcd(scale, duration) * duration;
    if (scale > MAX_RATIONAL_SCALE) return null;
  }
  const prepared = eligible.map((row) => {
    const base = BigInt(row.mediaStartMs) * scale;
    const rate = BigInt(row.mediaEndMs - row.mediaStartMs) * (scale / BigInt(row.wallEndMs - row.wallStartMs));
    const at = Math.max(row.wallStartMs, eligibleAtMs);
    return { row, base, rate, eligibleAtMs: at, start: base + rate * BigInt(at - row.wallStartMs) };
  }).sort((a, b) => a.start < b.start ? -1 : a.start > b.start ? 1 : 0);
  return { scale, segments: prepared };
}

/** Preserve original slopes and exact coverage. The fixed denominator is
 * shared by every interval, so union and threshold tests use integers only.
 * Sort eligible starts once and stop summing once the threshold is reached.
 */
function completedAt(durationMs: number, history: PreparedHistory, atMs: number): boolean {
  let end = BigInt(0);
  let covered = BigInt(0);
  const threshold = BigInt(durationMs) * BigInt(COMPLETION_PERCENT) * history.scale;
  for (const prepared of history.segments) {
    if (prepared.eligibleAtMs >= atMs) continue;
    const intervalEnd = prepared.base + prepared.rate * BigInt(Math.min(prepared.row.wallEndMs, atMs) - prepared.row.wallStartMs);
    const start = prepared.start > end ? prepared.start : end;
    if (intervalEnd > start) {
      covered += intervalEnd - start;
      end = intervalEnd;
      if (covered * BigInt(100) >= threshold) return true;
    }
  }
  return false;
}

    const [result]=await tx.$queryRawUnsafe('SELECT public.wp20_metric_completion_publish($1,$2,$3,$4,$5,$6::bigint,$7::integer,$8::jsonb) AS changed',lease.vendorId,lease.liveId,lease.versionId,lease.reportId,worker,source,count,JSON.stringify(rows));
    return result.changed;
  };
  const rowCount=async()=>Number((await tx.$queryRawUnsafe('SELECT count(*) AS n FROM "LiveMetricCompletionProjection" WHERE "reportId"=$1',lease.reportId))[0].n);
  assert.equal(await publish(facts,1,"wrong-worker"),0);
  assert.equal(await publish(facts,1,lease.workerId,revision+BigInt(1)),0);
  assert.equal(await rowCount(),0);
  const rejected=async action=>{
    await tx.$executeRawUnsafe("SAVEPOINT completion_guard_probe");
    let code=null;
    try {await action();}catch(error){code=error?.meta?.code??null;}
    await tx.$executeRawUnsafe("ROLLBACK TO SAVEPOINT completion_guard_probe");
    await tx.$executeRawUnsafe("RELEASE SAVEPOINT completion_guard_probe");
    assert.equal(code,"23514");
  };
  await rejected(()=>publish([{...facts[0],personId:"foreign-registration"}]));
  await rejected(()=>publish([{...facts[0],firstCompletedAtMs:+verifiedAt-1}]));
  await rejected(()=>publish([{...facts[0],firstCompletedAtMs:+verifiedAt+0.5}]));
  await rejected(()=>publish([{...facts[0],extra:true}]));
  await rejected(()=>publish(facts,2));
  assert.equal(await rowCount(),0);
  assert.equal(await publish(),1);
  assert.equal(await publish(),1);
  assert.equal(await rowCount(),1);
  await rejected(()=>publish([{...facts[0],firstCompletedAtMs:+verifiedAt+2}]));
  const [stored]=await tx.$queryRawUnsafe('SELECT rows FROM "LiveMetricCompletionProjection" WHERE "reportId"=$1',lease.reportId);
  assert.deepEqual(stored.rows,facts);
  // Use the real claim API and database clock; never bypass the report guard.
  await tx.$executeRawUnsafe("SAVEPOINT completion_expiry_probe");
  const expiredId = `${lease.reportId}-expiry`;
  const [created] = await tx.$queryRawUnsafe(`SELECT public.wp20_metric_report_create(
    $1,$2,$3,$4,"windowStartAt" + interval '1 millisecond',"windowEndAt","inputRevision","expiresAt") AS changed
    FROM "LiveMetricReport" WHERE id=$5`,lease.vendorId,lease.liveId,lease.versionId,expiredId,lease.reportId);
  assert.equal(created.changed,1);
  const claim = async (worker, milliseconds) => (await tx.$queryRawUnsafe(
    `SELECT public.wp20_metric_report_claim($1,$2,$3,$4,$5,clock_timestamp()+$6::integer*interval '1 millisecond') AS changed`,
    lease.vendorId,lease.liveId,lease.versionId,expiredId,worker,milliseconds))[0].changed;
  const publishExpired = async worker => (await tx.$queryRawUnsafe(
    'SELECT public.wp20_metric_completion_publish($1,$2,$3,$4,$5,$6::bigint,1,$7::jsonb) AS changed',
    lease.vendorId,lease.liveId,lease.versionId,expiredId,worker,revision,JSON.stringify(facts)))[0].changed;
  assert.equal(await claim("expired-worker",100),1);
  await tx.$executeRawUnsafe("DO $$ BEGIN PERFORM pg_sleep(0.15); END $$");
  assert.equal(await publishExpired("expired-worker"),0);
  assert.equal(Number((await tx.$queryRawUnsafe('SELECT count(*) AS n FROM "LiveMetricCompletionProjection" WHERE "reportId"=$1',expiredId))[0].n),0);
  assert.equal(await claim("replacement-worker",60000),1);
  assert.equal(await publishExpired("expired-worker"),0);
  assert.equal(await publishExpired("replacement-worker"),1);
  await tx.$executeRawUnsafe("ROLLBACK TO SAVEPOINT completion_expiry_probe");
  await tx.$executeRawUnsafe("RELEASE SAVEPOINT completion_expiry_probe");
  for(const role of ["anon","authenticated","service_role"]){
    const [rights]=await tx.$queryRawUnsafe(`SELECT has_function_privilege($1,'public.wp20_metric_completion_publish(text,text,text,text,text,bigint,integer,jsonb)','EXECUTE') AS allowed`,role);
    assert.equal(rights.allowed,false);
  }
