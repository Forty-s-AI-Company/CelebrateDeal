11:   limits = { rows: 1_000_000, pages: 2000 }) {
12:   if (!Number.isSafeInteger(limits.rows) || limits.rows < 1 || limits.rows > 1_000_000
13:     || !Number.isSafeInteger(limits.pages) || limits.pages < 1 || limits.pages > 2000) throw new Error("Invalid order audit limits");

15:     startAt: new Date(value.startAt), endAt: new Date(value.endAt) };

17:   return await db.$transaction(async rawTx => {
18:     await rawTx.$executeRaw(Prisma.sql`SET TRANSACTION READ ONLY`);
19:     await rawTx.$queryRaw(Prisma.sql`SELECT set_config('statement_timeout','2000',true)`);
20:     const started = performance.now();
21:     return withMetricOrderAuditContext(rawTx, scope, async tx => {
22:     const digest = createHash("sha256").update(JSON.stringify({ format: "order-source-audit-v1", ...scope }));
23:     const outcomes = { missing_payment: 0, missing_attribution: 0, other_purpose: 0, other_live: 0,
24:       invalid_source: 0, outside_cohort: 0, missing_registration: 0, identity_mismatch: 0, matched: 0 };
25:     let count = 0, cursor: string | null = null, revisions: string | undefined;
26:     for (let index = 0; index < limits.pages; index++) {
27:       // Context was read once; retain a conservative margin for DB/transaction overhead.
28:       if (performance.now() - started >= 20000) return { status: "unavailable" as const, reason: "limit_exceeded" as const };
29:       const page = await readMetricOrderAuditPage(tx, { ...scope, afterId: cursor, take: 500 });
30:       if (page.status !== "available") return page;
31:       const current = JSON.stringify([page.sourceRevision.toString(), page.orderRevision.toString(), page.registrationRevision.toString(), page.tenantRevision.toString()]);
32:       if (revisions !== undefined && current !== revisions) throw new Error("Order audit snapshot revision changed");
33:       revisions = current;
34:       count += page.rows.length;
35:       if (count > limits.rows) return { status: "unavailable" as const, reason: "limit_exceeded" as const };
36:       for (const row of page.rows) {
37:         outcomes[row.outcome]++;
38:         digest.update(JSON.stringify([row.orderId, row.outcome]) + "\n");
39:       }
40:       if (!page.hasMore) return { status: "enumerated" as const, coverage: "unknown" as const,

45:     }
46:     return { status: "unavailable" as const, reason: "limit_exceeded" as const };
