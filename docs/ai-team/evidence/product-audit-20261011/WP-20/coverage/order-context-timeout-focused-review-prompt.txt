18:   const context = cached?.key === scopeKey(input) ? cached.context : await readAuditContext(tx, input);

21:   const rows = await tx.$queryRaw<{ orderId: string; outcome: Outcome }[]>(Prisma.sql`
22:     SELECT o.id AS "orderId",CASE
23:       WHEN p.id IS NULL THEN 'missing_payment'
24:       WHEN coalesce(p.metadata->>'billingPurpose','')='' THEN 'missing_attribution'
25:       WHEN p.metadata->>'billingPurpose'<>'buyer_order' THEN 'other_purpose'
26:       WHEN coalesce(p.metadata->>'sourceLiveId','')='' OR coalesce(p.metadata->>'formSubmissionId','')='' THEN 'missing_attribution'
27:       WHEN p.metadata->>'sourceLiveId'<>${input.liveId} THEN 'other_live'
28:       WHEN s.id IS NULL OR s."liveId" IS DISTINCT FROM ${input.liveId} OR s."verificationStatus"<>'VERIFIED'
29:         OR f."vendorId" IS DISTINCT FROM ${input.vendorId} OR l."formId" IS DISTINCT FROM s."formId" OR s."verifiedAt" IS NULL THEN 'invalid_source'
30:       WHEN s."verifiedAt"<(${input.startAt}::timestamptz AT TIME ZONE 'UTC')
31:         OR s."verifiedAt">=(${input.endAt}::timestamptz AT TIME ZONE 'UTC') THEN 'outside_cohort'
32:       WHEN g.id IS NULL THEN 'missing_registration'
33:       WHEN g."verifiedAt" IS DISTINCT FROM (s."verifiedAt" AT TIME ZONE 'UTC') THEN 'identity_mismatch'
34:       ELSE 'matched' END AS outcome
35:     FROM "CommerceOrder" o
36:     LEFT JOIN "PaymentTransaction" p ON p.id=o."primaryPaymentTransactionId" AND p."vendorId"=o."vendorId"
37:     LEFT JOIN "FormSubmission" s ON s.id=p.metadata->>'formSubmissionId'
38:     LEFT JOIN "RegistrationForm" f ON f.id=s."formId" AND f."vendorId"=o."vendorId"
