import type { MetricScope } from "./live-core-metrics-read-model";

export type MetricEvidenceKind = "visits" | "registrations" | "viewing" | "orders";
export type MetricCoverageRange = { scope: MetricScope; kind: MetricEvidenceKind; startMs: number; endMs: number };
export type MetricCoverage = { status: "complete" } | { status: "unavailable";
  reason: "unknown_history" | "uncovered_window" | "evidence_gap" | "invalid_evidence" | "limit_exceeded" };
type Input = MetricCoverageRange & { covered: readonly MetricCoverageRange[] | null; gaps: readonly MetricCoverageRange[] | null };
const timestamp = (n: number) => Number.isSafeInteger(n) && n >= 0 && n <= 8_640_000_000_000_000;
const scopeValid = (s: MetricScope) => s && [s.vendorId, s.liveId, s.contentVersionId].every(v => typeof v === "string" && v.length > 0 && v.length <= 128)
  && (s.mode === "live" || s.mode === "replay");
const rangeValid = (r: MetricCoverageRange) => r && scopeValid(r.scope) && ["visits", "registrations", "viewing", "orders"].includes(r.kind)
  && timestamp(r.startMs) && timestamp(r.endMs) && r.startMs < r.endMs;

/** Evaluate durable, server-verified coverage evidence, never mere collection
 * enablement or the presence of rows. The adapter must load covered ranges AND
 * known losses in the same snapshot as facts. null means that audit is missing;
 * [] for gaps means an audited empty loss history, not an assumed default.
 * [S,T) gaps touching a boundary outside the requested window do not invalidate it.
 */
export function assessMetricCoverage(input: Input): MetricCoverage {
  if (!rangeValid(input)) return { status: "unavailable", reason: "invalid_evidence" };
  if (input.covered === null || input.gaps === null) return { status: "unavailable", reason: "unknown_history" };
  if (!Array.isArray(input.covered) || !Array.isArray(input.gaps)) return { status: "unavailable", reason: "invalid_evidence" };
  if (input.covered.length + input.gaps.length > 10000) return { status: "unavailable", reason: "limit_exceeded" };
  for (const rows of [input.covered, input.gaps]) for (const row of rows) {
    if (!rangeValid(row) || row.kind !== input.kind || row.scope.vendorId !== input.scope.vendorId
      || row.scope.liveId !== input.scope.liveId || row.scope.contentVersionId !== input.scope.contentVersionId
      || row.scope.mode !== input.scope.mode) return { status: "unavailable", reason: "invalid_evidence" };
  }
  if (input.gaps.some(row => row.startMs < input.endMs && row.endMs > input.startMs)) return { status: "unavailable", reason: "evidence_gap" };
  let end = input.startMs;
  for (const row of [...input.covered].sort((a, b) => a.startMs - b.startMs)) {
    if (row.endMs <= end) continue;
    if (row.startMs > end) break;
    end = row.endMs;
    if (end >= input.endMs) return { status: "complete" };
  }
  return { status: "unavailable", reason: "uncovered_window" };
}

type FailureReason = "limit_exceeded" | "lease_or_scope_unavailable" | "registration_outside_cohort" | "identity_changed";
class SnapshotUnavailable extends Error {
  constructor(readonly reason: FailureReason) { super(reason); }
}

/** Database I/O only: this does not run the CPU-heavy metric calculation.
 * All pages share one MVCC snapshot and one work budget. A failed page discards
 * the entire snapshot, while incomplete delivery remains explicit metadata.
 */
export async function loadMetricViewingSnapshot(db: Pick<PrismaClient, "$transaction">, lease: MetricReportLease,
  budget = new MetricReadBudget()): Promise<{ status: "available"; data: MetricViewingSnapshot } | { status: "unavailable"; reason: FailureReason }> {
  try {
    const data = await db.$transaction(async tx => {
      // This timeout limits each SQL statement; the transaction timeout bounds
      // all statements together. Neither changes the immutable evidence.
      await tx.$queryRaw(Prisma.sql`SELECT set_config('statement_timeout','5000',true)`);
      if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
      const context = await readMetricReportContext(tx, lease);
      if (context.status !== "available") throw new SnapshotUnavailable(context.reason);
      if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
      const audit = await readMetricCoverage(tx, lease, "visits");
      if (audit.status !== "available") throw new SnapshotUnavailable(audit.reason);
      if (!budget.retain(audit.rowCount)) throw new SnapshotUnavailable("limit_exceeded");
      const visitCoverage = audit.coverage;
      const result: MetricViewingSnapshot = { context: context.context, visitCoverage, visitLosses: [], visits: [], registrations: [], sessions: [], deliveryComplete: true };
      let afterLoss: Date | null = null;
      for (;;) {
        if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
        const page = await readMetricVisitLossPage(tx, { ...lease, after: afterLoss, take: 500 });
        if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
        if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");
        result.visitLosses.push(...page.rows);
        if (!page.hasMore) break;
        afterLoss = page.next;
      }
      // A proven gap is decisive even while the collection-start audit is unknown.
      if (result.visitLosses.length) result.visitCoverage = { status: "unavailable", reason: "evidence_gap" };
      let after: { id: string; visitedAt: Date } | null = null;
      for (;;) {
        if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
        const page = await readMetricVisitPage(tx, { ...lease, after, take: 500 });
        if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
        if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");
        result.visits.push(...page.rows);
        if (!page.hasMore) break;
        after = page.next;
      }
      let afterId: string | null = null;
      for (;;) {
        if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");
        const page = await readMetricRegistrationPage(tx, { ...lease, afterId, take: 100 });
        if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
        if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");
        for (const registration of page.rows) {
          if (!registration.identityCurrent) throw new SnapshotUnavailable("identity_changed");
          result.registrations.push(registration);
          await appendRegistrationSessions(tx, lease, registration.id, budget, result);
        }
        if (!page.hasMore) return result;
        afterId = page.nextId;
      }
    }, { isolationLevel: Prisma.TransactionIsolationLevel.RepeatableRead, timeout: 30000, maxWait: 5000 });
    return { status: "available", data };
  } catch (error) {
    if (error instanceof SnapshotUnavailable) return { status: "unavailable", reason: error.reason };
    // Infrastructure failures propagate to the job runner. They are never
    // converted into an empty cohort or a successfully calculated zero.
    throw error;
  }
}

async function appendRegistrationSessions(tx: Prisma.TransactionClient, lease: MetricReportLease, registrationId: string,
  budget: MetricReadBudget, result: MetricViewingSnapshot) {

    if (!budget.nextPage()) throw new SnapshotUnavailable("limit_exceeded");

    if (page.status !== "available") throw new SnapshotUnavailable(page.reason);
    if (!budget.retain(page.rows.length)) throw new SnapshotUnavailable("limit_exceeded");

      result.deliveryComplete &&= intervals.deliveryComplete;
    }
    if (!page.hasMore) return;
    afterId = page.nextId;
  }
}

    JOIN "Live" l ON l.id=g."liveId" AND l."vendorId"=g."vendorId"
    LEFT JOIN "FormSubmission" f ON f.id=g."submissionId"
    WHERE g."vendorId"=${input.vendorId} AND g."liveId"=${input.liveId} AND g."versionId"=${input.versionId}
      AND g."verifiedAt">=${report.windowStartAt} AND g."verifiedAt"<${report.windowEndAt} ${cursor}
    ORDER BY g.id ASC LIMIT ${input.take + 1}`);
  const page = rows.slice(0, input.take);
  return { status: "available" as const, rows: page, hasMore: rows.length > input.take, nextId: page.at(-1)?.id ?? input.afterId };
}

/** Shared by all sessions in one report, not reset per person/page. These are
 * work limits, not sampling or truncation rules. The caller must also bound the
 * database transaction timeout because a budget cannot cancel an active query.
 */
export class MetricReadBudget {
  private rows = 0;
  private pages = 0;
  private readonly started: number;
  constructor(private readonly limits = { rows: 1_000_000, pages: 4000, milliseconds: 30000 }, private readonly now = () => performance.now()) {
    if (!Number.isSafeInteger(limits.rows) || limits.rows < 1 || limits.rows > 1_000_000
      || !Number.isSafeInteger(limits.pages) || limits.pages < 1 || limits.pages > 4000
      || !Number.isSafeInteger(limits.milliseconds) || limits.milliseconds < 1 || limits.milliseconds > 30000) throw new Error("Invalid metric read budget");
    this.started = now();
  }
  active() { const elapsed = this.now() - this.started; return Number.isFinite(elapsed) && elapsed >= 0 && elapsed < this.limits.milliseconds; }
  nextPage() { if (!this.active() || this.pages >= this.limits.pages) return false; this.pages++; return true; }
  retain(count: number) {
    if (!Number.isSafeInteger(count) || count < 0 || !this.active() || count > this.limits.rows - this.rows) return false;
    this.rows += count; return true;
  }
}

export async function readMetricSessionIntervals(tx: Pick<Prisma.TransactionClient, "$queryRaw">,
  value: Omit<ReadRequest, "afterSequence" | "take">, budget: MetricReadBudget) {
  const rows: MetricIntervalRow[] = [];
  let afterSequence = -1;
  while (budget.nextPage()) {
    const page = await readMetricIntervalPage(tx, { ...value, afterSequence, take: 500 });
    if (page.status !== "available") return page;
    if (!budget.retain(page.rows.length)) break;
    rows.push(...page.rows);
    if (!page.hasMore) return { status: "available" as const, rows, registrationId: page.registrationId, deliveryComplete: page.deliveryComplete };
    afterSequence = page.nextSequence;
  }
  // Do not leak a prefix as if it represented all viewing for this session.
  return { status: "unavailable" as const, reason: "limit_exceeded" as const };
}

/** Read one indexed session/sequence page inside the worker's repeatable-read
 * snapshot. Never use OFFSET or an unbounded cohort query. Claim/finish happen
 * in separate read-committed transactions, as required by the write guards.
 */
export async function readMetricIntervalPage(tx: Pick<Prisma.TransactionClient, "$queryRaw">, value: ReadRequest) {
  const input = Request.parse(value);
  const [session] = await tx.$queryRaw<SessionState[]>(Prisma.sql`
    SELECT current_setting('transaction_isolation') AS isolation, s."registrationId",s."lastSequence",s.complete,s."closedAt"
    FROM "LiveMetricSession" s
    JOIN "LiveMetricReport" r ON r."vendorId"=s."vendorId" AND r."liveId"=s."liveId" AND r."versionId"=s."versionId"
    JOIN "LiveMetricVersion" v ON v.id=r."versionId" AND v."vendorId"=r."vendorId" AND v."liveId"=r."liveId"
    WHERE s.id=${input.sessionId} AND s."vendorId"=${input.vendorId} AND s."liveId"=${input.liveId} AND s."versionId"=${input.versionId}
      AND r.id=${input.reportId} AND r.status='running' AND r."leaseOwner"=${input.workerId}
      AND r."leaseExpiresAt">clock_timestamp() AND r."expiresAt">clock_timestamp()
      AND r."inputRevision"=v."sourceRevision"`);
  if (!session) return { status: "unavailable" as const, reason: "lease_or_scope_unavailable" as const };
  if (session.isolation !== "repeatable read") throw new Error("Metric reader requires repeatable-read snapshot");
  if (input.afterSequence > session.lastSequence) throw new Error("Metric cursor exceeds snapshot");
  const rows = await tx.$queryRaw<MetricIntervalRow[]>(Prisma.sql`
    SELECT sequence,"wallStartAt","wallEndAt","mediaStartMs","mediaEndMs" FROM "LiveMetricInterval"
    WHERE "sessionId"=${input.sessionId} AND "vendorId"=${input.vendorId} AND "liveId"=${input.liveId} AND "versionId"=${input.versionId}
      AND sequence>${input.afterSequence} ORDER BY sequence ASC LIMIT ${input.take + 1}`);
  if (rows.length > input.take + 1 || rows.some((row, index) => row.sequence !== input.afterSequence + index + 1)) throw new Error("Metric interval snapshot is inconsistent");
  const hasMore = rows.length > input.take;
  const page = rows.slice(0, input.take);
  const nextSequence = page.at(-1)?.sequence ?? input.afterSequence;
  if (!hasMore && nextSequence !== session.lastSequence) throw new Error("Metric interval snapshot is incomplete");
  return { status: "available" as const, registrationId: session.registrationId, rows: page, nextSequence, hasMore,
    // Complete pagination and complete delivery are different facts.
    deliveryComplete: session.complete && session.closedAt !== null };
}
