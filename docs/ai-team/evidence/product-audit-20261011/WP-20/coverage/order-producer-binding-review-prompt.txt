2: import { Prisma, type PrismaClient } from "@prisma/client";
3: import { auditMetricOrderSources } from "./live-metric-order-source-audit";
4: type Scope = Parameters<typeof auditMetricOrderSources>[1];
5: /** Private backend producer. Never accepts caller-supplied counts, proof or
6:  * completeness. Only an internally completed, tenant inventory audit with explicit exclusions
7:  * can publish; current revisions are checked again in the write transaction.
8:  */
9: export async function publishOrderCoverage(db: Pick<PrismaClient, "$transaction">, value: Scope) {
10:   const scope = { vendorId: value.vendorId, liveId: value.liveId, versionId: value.versionId,
11:     startAt: new Date(value.startAt), endAt: new Date(value.endAt) };
12:   const audit = await auditMetricOrderSources(db, scope);
13:   if (audit.status !== "enumerated") return { status: "unavailable" as const };
14:   // Exclude only positively classified out-of-scope orders. Missing or invalid
15:   // attribution/projections cannot silently disappear from the denominator.
16:   const accepted = audit.outcomes.matched + audit.outcomes.other_purpose
17:     + audit.outcomes.other_live + audit.outcomes.outside_cohort;
18:   if (accepted !== audit.count) return { status: "incomplete" as const };
19:   const proofId = createHash("sha256").update("order-coverage-v1:" + audit.digest).digest("hex");
20:   try {
21:   return await db.$transaction(async tx => {
22:     await tx.$queryRaw(Prisma.sql`SELECT set_config('statement_timeout','5000',true),set_config('lock_timeout','2000',true)`);
23:     const [version] = await tx.$queryRaw<{ sourceRevision: bigint }[]>(Prisma.sql`
24:       SELECT "sourceRevision" FROM "LiveMetricVersion" WHERE id=${scope.versionId}
25:         AND "vendorId"=${scope.vendorId} AND "liveId"=${scope.liveId} FOR SHARE`);
26:     if (!version || version.sourceRevision !== audit.sourceRevision) return { status: "source_changed" as const };
27:     const [source] = await tx.$queryRaw<{ registration: bigint; orders: bigint; tenant: bigint }[]>(Prisma.sql`
28:       SELECT public.wp20_metric_registration_revision(${scope.vendorId},${scope.liveId}) AS registration,
29:         public.wp20_metric_order_revision(${scope.vendorId},${scope.liveId}) AS orders,
30:         public.wp20_metric_order_tenant_revision(${scope.vendorId}) AS tenant`);
31:     if (!source || source.registration !== audit.registrationRevision
32:       || source.orders !== audit.orderRevision || source.tenant !== audit.tenantRevision) return { status: "source_changed" as const };
33:     await tx.$queryRaw(Prisma.sql`
34:       INSERT INTO "LiveMetricCoverageAudit" (id,"vendorId","liveId","versionId",kind,status,"startAt","endAt",
35:         "sourceRevision","orderRevision","orderTenantRevision","orderRegistrationRevision","orderCount","auditDigest")
36:       VALUES (${proofId},${scope.vendorId},${scope.liveId},${scope.versionId},'orders','covered',${scope.startAt},${scope.endAt},
37:         ${audit.sourceRevision},${audit.orderRevision},${audit.tenantRevision},${audit.registrationRevision},${audit.count},${audit.digest}) ON CONFLICT (id) DO NOTHING`);

40:   } catch (error) {

42:     // dedicated DB guard is a stale-proof result; other database errors propagate.
43:     if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2010"
44:       && error.meta?.code === "23514"
45:       && typeof error.meta?.message === "string"
46:       && error.meta.message.includes("order coverage source mismatch")) {
47:       return { status: "source_changed" as const };
48:     }
49:     throw error;
50:   }
51: }

7: const Kind = z.enum(["visits", "registrations", "viewing", "orders"]);
8: type AuditRow = { status: "covered" | "gap"; startAt: Date; endAt: Date };
9: 
10: /** Caller supplies the same RepeatableRead transaction used for report facts.
11:  * Old positive certificates expire on structural revision changes; known gaps
12:  * remain evidence of loss. No audit rows means unknown, never an audited zero.
13:  */
14: export async function readMetricCoverage(tx: Pick<Prisma.TransactionClient, "$queryRaw">,
15:   lease: MetricReportLease, value: z.infer<typeof Kind>) {
16:   const kind = Kind.parse(value);
17:   const context = await readMetricReportContext(tx, lease);
18:   if (context.status !== "available") return context;
19:   const rows = await tx.$queryRaw<AuditRow[]>(Prisma.sql`
20:     SELECT a.status,a."startAt",a."endAt" FROM "LiveMetricCoverageAudit" a
21:     JOIN "LiveMetricVersion" v ON v.id=a."versionId" AND v."vendorId"=a."vendorId" AND v."liveId"=a."liveId"
22:     WHERE a."vendorId"=${lease.vendorId} AND a."liveId"=${lease.liveId} AND a."versionId"=${lease.versionId}
23:       AND a.kind=${kind} AND (a.status='gap' OR (a."sourceRevision"=v."sourceRevision"
24:         AND (a.kind<>'registrations' OR (a."registrationCount" IS NOT NULL
25:           AND a."registrationRevision"=public.wp20_metric_registration_revision(a."vendorId",a."liveId")))
26:         AND (a.kind<>'orders' OR (a."orderCount" IS NOT NULL
27:           AND a."orderRevision"=public.wp20_metric_order_revision(a."vendorId",a."liveId")
28:           AND a."orderTenantRevision"=public.wp20_metric_order_tenant_revision(a."vendorId")
29:           AND a."orderRegistrationRevision"=public.wp20_metric_registration_revision(a."vendorId",a."liveId")))))
30:       AND a."startAt"<${context.context.windowEndAt} AND a."endAt">${context.context.windowStartAt}
31:     ORDER BY a."startAt",a."endAt",a.id LIMIT 10001`);
32:   if (rows.length > 10000) return { status: "unavailable" as const, reason: "limit_exceeded" as const };
33:   const scope = { vendorId: lease.vendorId, liveId: lease.liveId, contentVersionId: lease.versionId, mode: context.context.mode };
34:   const ranges = (status: AuditRow["status"]) => rows.filter(row => row.status === status)
35:     .map(row => ({ scope, kind, startMs: row.startAt.getTime(), endMs: row.endAt.getTime() }));
36:   return { status: "available" as const, rowCount: rows.length,
37:     coverage: assessMetricCoverage({ scope, kind, startMs: context.context.windowStartAt.getTime(),
38:       endMs: context.context.windowEndAt.getTime(), covered: rows.length ? ranges("covered") : null,

202:  SELECT "sourceRevision" INTO current_revision FROM public."LiveMetricVersion"
203:   WHERE id=version_id AND "vendorId"=vendor_id AND "liveId"=live_id FOR UPDATE;
204:  IF NOT FOUND OR current_revision<>source_revision THEN RETURN 0; END IF;
205:  SELECT * INTO report FROM public."LiveMetricReport" WHERE id=report_id
206:   AND "vendorId"=vendor_id AND "liveId"=live_id AND "versionId"=version_id FOR UPDATE;
207:  IF NOT FOUND OR report.status<>'running' OR report."leaseOwner" IS DISTINCT FROM worker_id
208:   OR report."leaseExpiresAt"<=clock_timestamp() OR report."expiresAt"<=clock_timestamp()
209:   OR report."inputRevision"<>source_revision
210:   OR report."inputOrderRevision" IS DISTINCT FROM public.wp20_metric_order_revision(vendor_id,live_id)
211:   OR report."inputRegistrationRevision" IS DISTINCT FROM public.wp20_metric_registration_revision(vendor_id,live_id)
212:   OR report."inputOrderTenantRevision" IS DISTINCT FROM public.wp20_metric_order_tenant_revision(vendor_id)
213:  THEN RETURN 0; END IF;
214:  SELECT count(*) INTO actual_cohort FROM public."LiveMetricRegistration"
215:   WHERE "vendorId"=vendor_id AND "liveId"=live_id AND "versionId"=version_id
216:     AND "verifiedAt">=report."windowStartAt" AND "verifiedAt"<report."windowEndAt";
217:  IF actual_cohort<>cohort_count THEN RAISE EXCEPTION 'completion cohort changed' USING ERRCODE='23514'; END IF;
218:  -- Scope-dependent membership remains under the version/report locks.
219:  IF EXISTS(
220:   SELECT 1 FROM jsonb_to_recordset(facts) AS f("personId" text,"firstCompletedAtMs" numeric)
221:   LEFT JOIN public."LiveMetricRegistration" g
222:    ON g.id=f."personId" AND g."vendorId"=vendor_id AND g."liveId"=live_id AND g."versionId"=version_id
223:     AND g."verifiedAt">=report."windowStartAt" AND g."verifiedAt"<report."windowEndAt"
224:     AND f."firstCompletedAtMs">=extract(epoch FROM g."verifiedAt")*1000
225:     AND f."firstCompletedAtMs"<extract(epoch FROM report."windowEndAt")*1000
226:   WHERE g.id IS NULL) THEN
227:   RAISE EXCEPTION 'completion fact outside cohort window' USING ERRCODE='23514';
228:  END IF;
229:  -- Validation may consume the rest of a short lease; do not publish after expiry.
230:  IF report."leaseExpiresAt"<=clock_timestamp() OR report."expiresAt"<=clock_timestamp() THEN RETURN 0; END IF;
231:  INSERT INTO public."LiveMetricCompletionProjection" ("reportId","sourceRevision","algorithmVersion","cohortCount","rowCount",rows)
232:  VALUES (report_id,source_revision,1,cohort_count,jsonb_array_length(facts),facts)
233:  ON CONFLICT ("reportId") DO NOTHING;
234:  IF NOT EXISTS(SELECT 1 FROM public."LiveMetricCompletionProjection" p WHERE p."reportId"=report_id
235:   AND p."sourceRevision"=source_revision AND p."cohortCount"=cohort_count AND p.rows=facts) THEN
236:   RAISE EXCEPTION 'completion projection conflict' USING ERRCODE='23514';
