11: export async function auditMetricRegistrationSources(db: Pick<PrismaClient, "$transaction">, scope: Scope,
12:   limits = { rows: 1_000_000, pages: 2000 }) {
13:   if (!Number.isSafeInteger(limits.rows) || limits.rows < 1 || limits.rows > 1_000_000
14:     || !Number.isSafeInteger(limits.pages) || limits.pages < 1 || limits.pages > 2000) throw new Error("Invalid audit limits");
15:   return db.$transaction(async tx => {
16:     await tx.$executeRaw(Prisma.sql`SET TRANSACTION READ ONLY`);
17:     await tx.$queryRaw(Prisma.sql`SELECT set_config('statement_timeout','5000',true)`);
18:     const started = performance.now();
19:     const digest = createHash("sha256").update(JSON.stringify({ format: "registration-source-audit-v2", vendorId: scope.vendorId, liveId: scope.liveId,
20:       versionId: scope.versionId, startAt: scope.startAt.toISOString(), endAt: scope.endAt.toISOString() }));
21:     const outcomes = { form_mismatch: 0, matched: 0, missing_capture: 0, missing_registration: 0, identity_mismatch: 0, other_version: 0 };
22:     let cursor: string | null = null, count = 0;
23:     let sourceRevision: bigint | undefined;
24:     let registrationRevision: bigint | undefined;
25:     for (let pageNumber = 0; pageNumber < limits.pages; pageNumber++) {
26:       if (performance.now() - started >= 23000) return { status: "unavailable" as const, reason: "limit_exceeded" as const };
27:       const page = await readRegistrationAuditPage(tx, { ...scope, take: 500, afterSubmissionId: cursor });
28:       if (page.status !== "available") return page;
29:       if (sourceRevision !== undefined && sourceRevision !== page.sourceRevision) throw new Error("Audit snapshot revision changed");
30:       if (registrationRevision !== undefined && registrationRevision !== page.registrationRevision) throw new Error("Audit registration revision changed");
31:       sourceRevision = page.sourceRevision;
32:       registrationRevision = page.registrationRevision;
33:       count += page.rows.length;
34:       if (count > limits.rows) return { status: "unavailable" as const, reason: "limit_exceeded" as const };
35:       for (const row of page.rows) {

38:       }
39:       if (!page.hasMore) {

41:           { sourceRevision, registrationRevision, started }, digest);

43:         return { status: "enumerated" as const, coverage: "unknown" as const, sourceRevision, registrationRevision,
44:           count, outcomes, projectionCount: reverse.count, projectionOutcomes: reverse.outcomes,
45:           digest: digest.update(JSON.stringify([sourceRevision.toString(), registrationRevision.toString()])).digest("hex") };
46:       }
47:       if (page.rows.length === 0 || page.nextSubmissionId === null || page.nextSubmissionId === cursor) throw new Error("Audit cursor did not advance");
48:       cursor = page.nextSubmissionId;
49:     }
50:     return { status: "unavailable" as const, reason: "limit_exceeded" as const };
51:   }, { isolationLevel: Prisma.TransactionIsolationLevel.RepeatableRead, timeout: 30000, maxWait: 5000 });
52: }
