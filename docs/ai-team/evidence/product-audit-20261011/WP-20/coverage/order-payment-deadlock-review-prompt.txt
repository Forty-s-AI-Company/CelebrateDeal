3: /** Recognize only database serialization/deadlock failures. Callers retain
4:  * their existing bounded attempt count and must restart the whole transaction.
5:  */
6: export function isDatabaseWriteConflict(error: unknown): boolean {
7:   if (error instanceof Prisma.PrismaClientKnownRequestError) {
8:     if (error.code === "P2034") return true;
9:     // Raw queries expose PostgreSQL SQLSTATE rather than Prisma's P2034.
10:     return error.code === "P2010" && (error.meta?.code === "40001" || error.meta?.code === "40P01");
11:   }
12:   // The installed connector can expose a real deadlock as UnknownRequestError.
13:   // Scope text classification to that driver class; arbitrary application errors,
14:   // lock timeouts, unique violations and validation failures are not retryable here.
15:   return error instanceof Prisma.PrismaClientUnknownRequestError
16:     && /(?:kind:\s*Deadlock\b|\bdeadlock detected\b)/iu.test(error.message);
17: }
