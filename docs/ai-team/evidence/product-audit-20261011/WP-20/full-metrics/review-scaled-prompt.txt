const timestamp = (value: number) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000;

function ratio(numerator: number, denominator: number, unit: "percent" | "seconds" = "percent"): MetricValue {
  if (!denominator) return { status: "not_applicable", reason: "zero_denominator" };
  const value = numerator / denominator * (unit === "percent" ? 100 : 1);
  return Number.isFinite(value) ? { status: "available", value, numerator, denominator, unit } : invalid();
}

/** Union actual exposure time across sessions/devices. Never fill gaps. */
function unionLength(intervals: readonly (readonly [number, number])[]): number {
  let end = 0;
  let total = 0;
  for (const [start, nextEnd] of [...intervals].sort((a, b) => a[0] - b[0])) {
    total += Math.max(0, nextEnd - Math.max(end, start));
    end = Math.max(end, nextEnd);
  }
  return total;
}

class InvalidMetricFacts extends Error {}
const requireFact = (condition: unknown) => { if (!condition) throw new InvalidMetricFacts(); };
type Cohort = Map<string, RegistrationFact>;

function validateScopeAndWindow(input: CoreMetricsInput) {
  requireFact(input && scopeValid(input.scope) && timestamp(input.startMs) && timestamp(input.endMs) && input.startMs < input.endMs);
  // Fail closed on mixed scopes; silently dropping rows would change denominators.
  for (const rows of [input.visits, input.registrations, input.viewing, input.completions, input.orders]) {
    requireFact(rows === null || (Array.isArray(rows) && Array.from(rows).every((row) => row && sameScope(row.scope, input.scope))));
  }
}

function buildCohort(input: CoreMetricsInput): Cohort {
  const registrations: Cohort = new Map();
  for (const row of input.registrations!) {
    requireFact(id(row.personId) && timestamp(row.verifiedAtMs) && (row.browserId === null || id(row.browserId)));
    const previous = registrations.get(row.personId);
    requireFact(!previous || (previous.verifiedAtMs === row.verifiedAtMs && previous.browserId === row.browserId));
    registrations.set(row.personId, row);
  }
  return new Map([...registrations].filter(([, row]) => row.verifiedAtMs >= input.startMs && row.verifiedAtMs < input.endMs));
}

function registrationMetrics(input: CoreMetricsInput, cohort: Cohort, result: CoreMetrics) {
  if (input.visits === null) return;
  const visits = new Map<string, number>();
  for (const row of input.visits) {
    requireFact(id(row.browserId) && timestamp(row.visitedAtMs));
    if (row.visitedAtMs >= input.startMs && row.visitedAtMs < input.endMs) visits.set(row.browserId, Math.min(visits.get(row.browserId) ?? Infinity, row.visitedAtMs));
  }
  const converted = new Set<string>();
  let unattributed = 0;
  for (const row of cohort.values()) {
    const visit = row.browserId === null ? undefined : visits.get(row.browserId);
    if (visit !== undefined && row.verifiedAtMs >= visit) converted.add(row.browserId!);
    else unattributed++;
  }
  result.registrationRate = ratio(converted.size, visits.size);
  result.unattributedRegistrations = unattributed;
}

function collectSegments(input: CoreMetricsInput, cohort: Cohort): Map<string, ViewingFact[]> {
  const segments = new Map<string, ViewingFact[]>();
  for (const row of input.viewing!) {
    requireFact(id(row.personId) && timestamp(row.wallStartMs) && timestamp(row.wallEndMs) && row.wallEndMs > row.wallStartMs);
    requireFact(Number.isSafeInteger(row.mediaStartMs) && Number.isSafeInteger(row.mediaEndMs) && row.mediaStartMs >= 0 && row.mediaEndMs > row.mediaStartMs);
    if (input.durationMs !== null && input.durationMs > 0) requireFact(row.mediaEndMs <= input.durationMs);
    const registration = cohort.get(row.personId);
    if (!registration || row.wallEndMs <= registration.verifiedAtMs || row.wallStartMs >= input.endMs) continue;
    const rows = segments.get(row.personId) ?? [];
    rows.push(row);
    segments.set(row.personId, rows);
  }
  return segments;
}

function viewingMetrics(input: CoreMetricsInput, cohort: Cohort, result: CoreMetrics): Map<string, number> {
  const segments = collectSegments(input, cohort);
  const completed = new Map<string, number>();
  let completionLimit = false;
  let wallMs = 0;
  for (const [person, rows] of segments) {
    const verifiedAt = cohort.get(person)!.verifiedAtMs;
    wallMs += unionLength(rows.map((row) => [Math.max(row.wallStartMs, verifiedAt, input.startMs), Math.min(row.wallEndMs, input.endMs)] as const));
    if (input.durationMs === null) continue;
    const completion = firstCompletionTime(input.durationMs, rows, { startMs: Math.max(verifiedAt, input.startMs), endMs: input.endMs - 1 });
    if (completion.status === "unavailable" && completion.reason === "limit_exceeded") {
      completionLimit = true;
      continue;
    }
    requireFact(completion.status === "available");
    if (completion.status === "available" && completion.firstCompletedAtMs !== null) completed.set(person, completion.firstCompletedAtMs);
  }
  result.attendanceRate = ratio(segments.size, cohort.size);
  result.averageViewingSeconds = ratio(wallMs / 1000, segments.size, "seconds");
  result.completionRate = input.durationMs === null ? missing() : completionLimit ? { status: "unavailable", reason: "limit_exceeded" } : ratio(completed.size, segments.size);
  return completed;
}

function orderMetrics(input: CoreMetricsInput, cohort: Cohort, completed: Map<string, number>, result: CoreMetrics) {
  if (result.completionRate.status === "unavailable") {
    result.postCompletionOrderRate = { ...result.completionRate };
    return;
  }
  if (input.completions === null || input.orders === null || input.durationMs === null) return;
  const firstCompletion = new Map<string, number>();
  for (const row of input.completions) {
    requireFact(id(row.personId) && timestamp(row.firstCompletedAtMs));
    const previous = firstCompletion.get(row.personId);
    requireFact(previous === undefined || previous === row.firstCompletedAtMs);
    firstCompletion.set(row.personId, row.firstCompletedAtMs);
  }
  for (const row of input.orders) requireFact(id(row.personId) && timestamp(row.establishedAtMs) && typeof row.cancelled === "boolean");
  // Reconcile durable first-crossing facts with independently derived viewing
  // evidence. [] is known-empty, so a contradiction is invalid, not missing.
  for (const [person, derivedAt] of completed) {
    requireFact(firstCompletion.get(person) === derivedAt);
  }
  for (const [person, at] of firstCompletion) {
    if (!cohort.has(person) || at >= input.endMs) continue;
    requireFact(at >= cohort.get(person)!.verifiedAtMs && completed.get(person) === at);
  }
  const ordered = new Set<string>();
  for (const row of input.orders) {
    if (!completed.has(row.personId) || row.establishedAtMs >= input.endMs) continue;
    const eligibility = evaluatePostCompletionOrder(firstCompletion.get(row.personId)!, row);
    if (eligibility.status === "available" && eligibility.qualifies) ordered.add(row.personId);
  }
  result.postCompletionOrderRate = ratio(ordered.size, completed.size);
  // The last millisecond of the inclusive seven-day window must be observed.
  result.provisionalConversions = [...completed.keys()].some((person) => input.endMs <= firstCompletion.get(person)! + POST_COMPLETION_ORDER_WINDOW_MS);
}

export function calculateCoreMetrics(input: CoreMetricsInput): CoreMetrics {
  const result: CoreMetrics = {
    registrationRate: missing(), attendanceRate: missing(), completionRate: missing(),
    averageViewingSeconds: missing(), postCompletionOrderRate: missing(),
    unattributedRegistrations: null, provisionalConversions: null,
  };
  try {
    validateScopeAndWindow(input);
    if (input.registrations === null) return result;
    const cohort = buildCohort(input);
    registrationMetrics(input, cohort, result);
    if (input.viewing === null) return result;
    requireFact(input.durationMs === null || (Number.isSafeInteger(input.durationMs) && input.durationMs > 0));
    const completed = viewingMetrics(input, cohort, result);
    orderMetrics(input, cohort, completed, result);
    return result;
  } catch (error) {
    if (!(error instanceof InvalidMetricFacts)) throw error;
    return { registrationRate: invalid(), attendanceRate: invalid(), completionRate: invalid(), averageViewingSeconds: invalid(), postCompletionOrderRate: invalid(), unattributedRegistrations: null, provisionalConversions: null };
  }
}

/** These are bounded exposure samples, not proof of human attention. A caller
 * must reset on seek, buffering, pause, source/version changes and rate changes.
 * Monotonic elapsed time is later anchored to server time by ingestion.
 */
export type PlaybackSample = Readonly<{
  elapsedMs: number; mediaMs: number; rate: number; playing: boolean;
}>;
export type PlaybackSegment = Readonly<{
  wallStartMs: number; wallEndMs: number; mediaStartMs: number; mediaEndMs: number;
}>;
export const MAX_PLAYBACK_SAMPLE_GAP_MS = 5000;
export const MAX_COMPLETION_SEGMENTS = 10000;

function validSample(sample: PlaybackSample): boolean {
  return !!sample && Number.isSafeInteger(sample.elapsedMs) && sample.elapsedMs >= 0
    && Number.isSafeInteger(sample.mediaMs) && sample.mediaMs >= 0
    && Number.isFinite(sample.rate) && sample.rate >= 0.25 && sample.rate <= 4
    && typeof sample.playing === "boolean";
}

/** A discontinuity returns no interval. Do not connect endpoints across gaps.
 * Integer millisecond samples allow one millisecond of quantization error.
 * This check bounds accidental seek jumps; the server still validates batches.
 */
export function playbackSegment(previous: PlaybackSample | null, current: PlaybackSample): PlaybackSegment | null {
  if (!previous || !validSample(previous) || !validSample(current) || !previous.playing || !current.playing || previous.rate !== current.rate) return null;
  const elapsed = current.elapsedMs - previous.elapsedMs;
  const media = current.mediaMs - previous.mediaMs;
  if (elapsed <= 0 || elapsed > MAX_PLAYBACK_SAMPLE_GAP_MS || media <= 0 || Math.abs(media - elapsed * current.rate) > 1) return null;
  return { wallStartMs: previous.elapsedMs, wallEndMs: current.elapsedMs, mediaStartMs: previous.mediaMs, mediaEndMs: current.mediaMs };
}

/** Derive the earliest integer millisecond at which unique coverage reached
 * 90%. Segments must be validated constant-rate exposure for one person/version.
 * Sorting by arrival order is incorrect for delayed and overlapping devices;
 * instead use a monotone coverage predicate over playback time.
 */
export function firstCompletionTime(durationMs: number | null, segments: readonly PlaybackSegment[] | null, window: Readonly<{ startMs: number; endMs: number }> = { startMs: 0, endMs: 8_640_000_000_000_000 }):
  | { status: "unavailable"; reason: "missing_data" | "invalid_data" | "limit_exceeded" }
  | { status: "available"; firstCompletedAtMs: number | null } {
  if (durationMs === null || segments === null) return { status: "unavailable", reason: "missing_data" };
  if (!Array.isArray(segments) || !Number.isSafeInteger(durationMs) || durationMs <= 0) return { status: "unavailable", reason: "invalid_data" };
  if (![window.startMs, window.endMs].every((value) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000) || window.startMs > window.endMs) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length > MAX_COMPLETION_SEGMENTS) return { status: "unavailable", reason: "limit_exceeded" };
  if (Array.from(segments).some((row) => !validSegment(row, durationMs))) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length === 0) return { status: "available", firstCompletedAtMs: null };
  let low = Infinity;
  let high = 0;
  for (const row of segments) {
    low = Math.min(low, row.wallStartMs);
    high = Math.max(high, row.wallEndMs);
  }
  low = Math.max(low, window.startMs);
  high = Math.min(high, window.endMs);
  const prepared = prepareSegments(segments, window.startMs);
  if (!prepared) return { status: "unavailable", reason: "limit_exceeded" };
  if (low > high || !completedAt(durationMs, prepared, high)) return { status: "available", firstCompletedAtMs: null };
  // At most 53 iterations for safe integer time. No per-millisecond scan and
  // no invented first-completion timestamp from receipt/arrival order.
  while (low < high) {
    const middle = low + Math.floor((high - low) / 2);
    if (completedAt(durationMs, prepared, middle)) high = middle;
    else low = middle + 1;
  }
  return { status: "available", firstCompletedAtMs: low };
}

function validSegment(row: PlaybackSegment, durationMs: number): boolean {
  return !!row && [row.wallStartMs, row.wallEndMs, row.mediaStartMs, row.mediaEndMs].every((value) => Number.isSafeInteger(value) && value >= 0)
    && row.wallEndMs <= 8_640_000_000_000_000 && row.wallEndMs > row.wallStartMs
    && row.mediaEndMs > row.mediaStartMs && row.mediaEndMs <= durationMs;
}

function gcd(a: bigint, b: bigint): bigint {
  while (b !== BigInt(0)) { const next = a % b; a = b; b = next; }
  return a;
}
type PreparedSegment = { row: PlaybackSegment; start: bigint; base: bigint; rate: bigint; eligibleAtMs: number };
type PreparedHistory = { scale: bigint; segments: PreparedSegment[] };
const MAX_RATIONAL_SCALE = BigInt(1) << BigInt(16384);

function prepareSegments(segments: readonly PlaybackSegment[], eligibleAtMs: number): PreparedHistory | null {
  const eligible = segments.filter((row) => row.wallEndMs > eligibleAtMs);
  // One common exact scale avoids huge repeated fraction reductions inside
  // binary search. Bound its size before allocating scaled per-segment values.
  let scale = BigInt(1);
  for (const row of eligible) {
    const duration = BigInt(row.wallEndMs - row.wallStartMs);
    scale = scale / gcd(scale, duration) * duration;
    if (scale > MAX_RATIONAL_SCALE) return null;
  }
  const prepared = eligible.map((row) => {
    const base = BigInt(row.mediaStartMs) * scale;
    const rate = BigInt(row.mediaEndMs - row.mediaStartMs) * (scale / BigInt(row.wallEndMs - row.wallStartMs));
    const at = Math.max(row.wallStartMs, eligibleAtMs);
    return { row, base, rate, eligibleAtMs: at, start: base + rate * BigInt(at - row.wallStartMs) };
  }).sort((a, b) => a.start < b.start ? -1 : a.start > b.start ? 1 : 0);
  return { scale, segments: prepared };
}

/** Preserve original slopes and exact coverage. The fixed denominator is
 * shared by every interval, so union and threshold tests use integers only.
 * Sort eligible starts once and stop summing once the threshold is reached.
 */
function completedAt(durationMs: number, history: PreparedHistory, atMs: number): boolean {
  let end = BigInt(0);
  let covered = BigInt(0);
  const threshold = BigInt(durationMs) * BigInt(COMPLETION_PERCENT) * history.scale;
  for (const prepared of history.segments) {
    if (prepared.eligibleAtMs >= atMs) continue;
    const intervalEnd = prepared.base + prepared.rate * BigInt(Math.min(prepared.row.wallEndMs, atMs) - prepared.row.wallStartMs);
    const start = prepared.start > end ? prepared.start : end;
    if (intervalEnd > start) {
      covered += intervalEnd - start;
      end = intervalEnd;
      if (covered * BigInt(100) >= threshold) return true;
    }
  }
  return false;
}

/** Approved WP20 rules only. Callers must supply verified, same-person,
 * same-tenant, same-session/content-version facts. This module does not join
 * identities, infer missing events, or decide any metric denominator.
 */
export const COMPLETION_PERCENT = 90;
export const POST_COMPLETION_ORDER_WINDOW_MS = 7 * 24 * 60 * 60 * 1000;

type Unavailable = { status: "unavailable"; reason: "missing_data" | "invalid_data" };
export type MediaInterval = Readonly<{ startMs: number; endMs: number }>;

/** Intervals describe actually played media positions, not seek endpoints.
 * null means incomplete/unknown evidence; [] means a known empty history.
 * Replaying the same positions cannot increase unique coverage.
 */
export function evaluateVideoCompletion(
  durationMs: number | null,
  intervals: readonly MediaInterval[] | null,
): Unavailable | { status: "available"; coveredMs: number; completed: boolean } {
  if (durationMs == null || intervals == null) return { status: "unavailable", reason: "missing_data" };
  if (!Number.isSafeInteger(durationMs) || durationMs <= 0 || !Array.isArray(intervals) || Array.from(intervals).some((interval) =>
    !interval || typeof interval !== "object"
    || !Number.isSafeInteger(interval.startMs) || !Number.isSafeInteger(interval.endMs)
    || interval.startMs < 0 || interval.endMs < interval.startMs || interval.endMs > durationMs)) {
    return { status: "unavailable", reason: "invalid_data" };
  }
  const sorted = [...intervals].sort((a, b) => a.startMs - b.startMs);
  let coveredMs = 0;
  let end = 0;
  for (const interval of sorted) {
    coveredMs += Math.max(0, interval.endMs - Math.max(end, interval.startMs));
    end = Math.max(end, interval.endMs);
  }
  // Integer arithmetic avoids rounding a below-threshold viewer up to 90%.
  const completed = BigInt(coveredMs) * BigInt(100) >= BigInt(durationMs) * BigInt(COMPLETION_PERCENT);
  return { status: "available", coveredMs, completed };
}

export type EstablishedOrderFact = Readonly<{
  establishedAtMs: number | null;
  cancelled: boolean | null;
}>;

/** Completion time must come from verified first threshold-crossing evidence;
 * a total coverage number alone cannot establish when completion occurred.
 * The 7-day boundary is inclusive, using elapsed UTC milliseconds, not dates.
 * This is order eligibility, not paid conversion or mature cohort reporting.
 */
export function evaluatePostCompletionOrder(
  firstCompletedAtMs: number | null,
  order: EstablishedOrderFact,
): Unavailable | { status: "available"; qualifies: boolean } {
  if (firstCompletedAtMs == null || order == null || order.establishedAtMs == null || order.cancelled == null) {
    return { status: "unavailable", reason: "missing_data" };
  }
  if (typeof order.cancelled !== "boolean" || ![firstCompletedAtMs, order.establishedAtMs].every((value) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000)) {
    return { status: "unavailable", reason: "invalid_data" };
  }
  const elapsed = order.establishedAtMs - firstCompletedAtMs;
  return { status: "available", qualifies: !order.cancelled && elapsed >= 0 && elapsed <= POST_COMPLETION_ORDER_WINDOW_MS };
}

var COMPLETION_PERCENT = 90;
var POST_COMPLETION_ORDER_WINDOW_MS = 7 * 24 * 60 * 60 * 1e3;

// src/lib/live-viewing-evidence.ts
var MAX_PLAYBACK_SAMPLE_GAP_MS = 5e3;
var MAX_COMPLETION_SEGMENTS = 1e4;
function validSample(sample) {
  return !!sample && Number.isSafeInteger(sample.elapsedMs) && sample.elapsedMs >= 0 && Number.isSafeInteger(sample.mediaMs) && sample.mediaMs >= 0 && Number.isFinite(sample.rate) && sample.rate >= 0.25 && sample.rate <= 4 && typeof sample.playing === "boolean";
}
function playbackSegment(previous, current) {
  if (!previous || !validSample(previous) || !validSample(current) || !previous.playing || !current.playing || previous.rate !== current.rate) return null;
  const elapsed = current.elapsedMs - previous.elapsedMs;
  const media = current.mediaMs - previous.mediaMs;
  if (elapsed <= 0 || elapsed > MAX_PLAYBACK_SAMPLE_GAP_MS || media <= 0 || Math.abs(media - elapsed * current.rate) > 1) return null;
  return { wallStartMs: previous.elapsedMs, wallEndMs: current.elapsedMs, mediaStartMs: previous.mediaMs, mediaEndMs: current.mediaMs };
}
function firstCompletionTime(durationMs, segments, window = { startMs: 0, endMs: 864e13 }) {
  if (durationMs === null || segments === null) return { status: "unavailable", reason: "missing_data" };
  if (!Array.isArray(segments) || !Number.isSafeInteger(durationMs) || durationMs <= 0) return { status: "unavailable", reason: "invalid_data" };
  if (![window.startMs, window.endMs].every((value) => Number.isSafeInteger(value) && value >= 0 && value <= 864e13) || window.startMs > window.endMs) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length > MAX_COMPLETION_SEGMENTS) return { status: "unavailable", reason: "limit_exceeded" };
  if (Array.from(segments).some((row) => !validSegment(row, durationMs))) return { status: "unavailable", reason: "invalid_data" };
  if (segments.length === 0) return { status: "available", firstCompletedAtMs: null };
  let low = Infinity;
  let high = 0;
  for (const row of segments) {
    low = Math.min(low, row.wallStartMs);
    high = Math.max(high, row.wallEndMs);
  }
  low = Math.max(low, window.startMs);
  high = Math.min(high, window.endMs);
  if (low > high || !completedAt(durationMs, segments, high, window.startMs)) return { status: "available", firstCompletedAtMs: null };
  while (low < high) {
    const middle = low + Math.floor((high - low) / 2);
    if (completedAt(durationMs, segments, middle, window.startMs)) high = middle;
    else low = middle + 1;
  }
  return { status: "available", firstCompletedAtMs: low };
}
function validSegment(row, durationMs) {
  return !!row && [row.wallStartMs, row.wallEndMs, row.mediaStartMs, row.mediaEndMs].every((value) => Number.isSafeInteger(value) && value >= 0) && row.wallEndMs <= 864e13 && row.wallEndMs > row.wallStartMs && row.mediaEndMs > row.mediaStartMs && row.mediaEndMs <= durationMs;
}
var compare = (a, b) => a.n * b.d - b.n * a.d;
function gcd(a, b) {
  while (b !== BigInt(0)) {
    const next = a % b;
    a = b;
    b = next;
  }
  return a;
}
function add(a, b) {
  const shared = gcd(a.d, b.d);
  const n = a.n * (b.d / shared) + b.n * (a.d / shared);
  const d = a.d * (b.d / shared);
  const divisor = gcd(n, d);
  return { n: n / divisor, d: d / divisor };
}
function mediaAt(row, atMs) {
  const d = BigInt(row.wallEndMs - row.wallStartMs);
  return { n: BigInt(row.mediaStartMs) * d + BigInt(row.mediaEndMs - row.mediaStartMs) * BigInt(atMs - row.wallStartMs), d };
}
function completedAt(durationMs, segments, atMs, eligibleAtMs) {
  const intervals = segments.filter((row) => Math.max(row.wallStartMs, eligibleAtMs) < Math.min(row.wallEndMs, atMs)).map((row) => ({
    start: mediaAt(row, Math.max(row.wallStartMs, eligibleAtMs)),
    end: mediaAt(row, Math.min(row.wallEndMs, atMs))
  })).sort((a, b) => {
    const order = compare(a.start, b.start);
    return order < BigInt(0) ? -1 : order > BigInt(0) ? 1 : 0;
  });
  let end = { n: BigInt(0), d: BigInt(1) };
  let covered = { n: BigInt(0), d: BigInt(1) };
  for (const interval of intervals) {
    const start = compare(interval.start, end) > BigInt(0) ? interval.start : end;
    if (compare(interval.end, start) > BigInt(0)) {
      covered = add(covered, { n: interval.end.n * start.d - start.n * interval.end.d, d: interval.end.d * start.d });
      end = interval.end;
    }
  }
  return covered.n * BigInt(100) >= BigInt(durationMs) * BigInt(COMPLETION_PERCENT) * covered.d;
}
