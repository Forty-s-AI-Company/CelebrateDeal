## 共通候選規則

採單一 vendor、場次、內容版本為範圍；開始 S 與截止 T 由商家明確選取，時間以 UTC 計算、介面顯示時區。直播與重播分開，不跨版本拼接完看。沒有完整版本時長的直播，完看相關數值維持不可計算，直到錄影版本與時長確定。

候選個人識別為伺服器驗證後的報名者識別，不使用電子郵件明文作分析 key；匿名瀏覽只用短期第一方瀏覽器識別，不能宣稱跨裝置唯一人數。是否可採用此追蹤方式、保存期限及告知內容仍需確認；不得推定 D-5 已解決。無法連結的資料列為未歸屬，不強行合併。只有已知且完整的零筆資料可顯示 0；分母為零顯示不適用，缺資料顯示不可計算並列原因。

觀看事件只能支持播放暴露量，不能證明人有專注觀看。事件需伺服器驗證場次、租戶、身分與時間，具備冪等 key；重送不得重複計算。離線、缺失或異常間隔不補猜。

import { evaluatePostCompletionOrder, evaluateVideoCompletion, POST_COMPLETION_ORDER_WINDOW_MS } from "./live-metric-contract";

/** Pure reporting contract. The persistence adapter must provide complete,
 * verified facts for one tenant/live/content version and viewing mode.
 * null means unknown history; [] means an audited, complete empty history.
 * IDs are opaque internal references, never email addresses or session counts.
 */
export type MetricScope = Readonly<{
  vendorId: string;
  liveId: string;
  contentVersionId: string;
  mode: "live" | "replay";
}>;
type Scoped = Readonly<{ scope: MetricScope }>;
export type VisitFact = Scoped & Readonly<{ browserId: string; visitedAtMs: number }>;
export type RegistrationFact = Scoped & Readonly<{
  personId: string; browserId: string | null; verifiedAtMs: number;
}>;
/** Each segment has a constant playback rate; seek/pause/buffering split it.
 * Wall timestamps are validated by ingestion, not taken on trust from clients.
 */
export type ViewingFact = Scoped & Readonly<{
  personId: string;
  wallStartMs: number; wallEndMs: number;
  mediaStartMs: number; mediaEndMs: number;
}>;
export type CompletionFact = Scoped & Readonly<{ personId: string; firstCompletedAtMs: number }>;
export type OrderFact = Scoped & Readonly<{
  personId: string; establishedAtMs: number; cancelled: boolean;
}>;
export type CoreMetricsInput = Readonly<{
  scope: MetricScope;
  startMs: number; endMs: number; durationMs: number | null;
  visits: readonly VisitFact[] | null;
  registrations: readonly RegistrationFact[] | null;
  viewing: readonly ViewingFact[] | null;
  completions: readonly CompletionFact[] | null;
  orders: readonly OrderFact[] | null;
}>;
export type MetricValue =
  | { status: "unavailable"; reason: "missing_data" | "invalid_data" }
  | { status: "not_applicable"; reason: "zero_denominator" }
  | { status: "available"; value: number; numerator: number; denominator: number; unit: "percent" | "seconds" };
export type CoreMetrics = {
  registrationRate: MetricValue;
  attendanceRate: MetricValue;
  completionRate: MetricValue;
  averageViewingSeconds: MetricValue;
  postCompletionOrderRate: MetricValue;
  unattributedRegistrations: number | null;
  provisionalConversions: boolean | null;
};

const missing = (): MetricValue => ({ status: "unavailable", reason: "missing_data" });
const invalid = (): MetricValue => ({ status: "unavailable", reason: "invalid_data" });
const timestamp = (value: number) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000;
const id = (value: string) => typeof value === "string" && value.length > 0;
const scopeValid = (scope: MetricScope) => scope && id(scope.vendorId) && id(scope.liveId) && id(scope.contentVersionId) && (scope.mode === "live" || scope.mode === "replay");
const sameScope = (a: MetricScope, b: MetricScope) => scopeValid(a) && a.vendorId === b.vendorId && a.liveId === b.liveId && a.contentVersionId === b.contentVersionId && a.mode === b.mode;

function ratio(numerator: number, denominator: number, unit: "percent" | "seconds" = "percent"): MetricValue {
  if (!denominator) return { status: "not_applicable", reason: "zero_denominator" };
  const value = numerator / denominator * (unit === "percent" ? 100 : 1);
  return Number.isFinite(value) ? { status: "available", value, numerator, denominator, unit } : invalid();
}

/** Union actual exposure time across sessions/devices. Never fill gaps. */
function unionLength(intervals: readonly (readonly [number, number])[]): number {
  let end = 0;
  let total = 0;
  for (const [start, nextEnd] of [...intervals].sort((a, b) => a[0] - b[0])) {
    total += Math.max(0, nextEnd - Math.max(end, start));
    end = Math.max(end, nextEnd);
  }
  return total;
}

export function calculateCoreMetrics(input: CoreMetricsInput): CoreMetrics {
  const result: CoreMetrics = {
    registrationRate: missing(), attendanceRate: missing(), completionRate: missing(),
    averageViewingSeconds: missing(), postCompletionOrderRate: missing(),
    unattributedRegistrations: null, provisionalConversions: null,
  };
  const reject = (): CoreMetrics => ({ ...result, registrationRate: invalid(), attendanceRate: invalid(), completionRate: invalid(), averageViewingSeconds: invalid(), postCompletionOrderRate: invalid() });
  if (!input || !scopeValid(input.scope) || !timestamp(input.startMs) || !timestamp(input.endMs) || input.startMs >= input.endMs) return reject();
  const { startMs, endMs } = input;
  // Cross-scope facts invalidate the batch instead of silently leaking or
  // discarding rows and reporting an apparently complete denominator.
  for (const rows of [input.visits, input.registrations, input.viewing, input.completions, input.orders]) {
    if (rows !== null && (!Array.isArray(rows) || Array.from(rows).some((row) => !row || !sameScope(row.scope, input.scope)))) return reject();
  }
  if (input.registrations === null) return result;
  const registrations = new Map<string, RegistrationFact>();
  for (const row of input.registrations) {
    if (!id(row.personId) || !timestamp(row.verifiedAtMs) || (row.browserId !== null && !id(row.browserId))) return reject();
    const previous = registrations.get(row.personId);
    // Exact retransmission is harmless; conflicting identity evidence isn't.
    if (previous && (previous.verifiedAtMs !== row.verifiedAtMs || previous.browserId !== row.browserId)) return reject();
    registrations.set(row.personId, row);
  }
  const cohort = new Map([...registrations].filter(([, row]) => row.verifiedAtMs >= startMs && row.verifiedAtMs < endMs));
  if (input.visits !== null) {
    const visits = new Map<string, number>();
    for (const row of input.visits) {
      if (!id(row.browserId) || !timestamp(row.visitedAtMs)) return reject();
      if (row.visitedAtMs >= startMs && row.visitedAtMs < endMs) visits.set(row.browserId, Math.min(visits.get(row.browserId) ?? Infinity, row.visitedAtMs));
    }
    const converted = new Set<string>();
    let unattributed = 0;
    for (const row of cohort.values()) {
      const visit = row.browserId === null ? undefined : visits.get(row.browserId);
      if (visit !== undefined && row.verifiedAtMs >= visit) converted.add(row.browserId!);
      else unattributed++;
    }
    result.registrationRate = ratio(converted.size, visits.size);
    result.unattributedRegistrations = unattributed;
  }
  if (input.viewing === null) return result;
  const segments = new Map<string, ViewingFact[]>();
  for (const row of input.viewing) {
    if (!id(row.personId) || !timestamp(row.wallStartMs) || !timestamp(row.wallEndMs) || row.wallEndMs <= row.wallStartMs
      || !Number.isSafeInteger(row.mediaStartMs) || !Number.isSafeInteger(row.mediaEndMs) || row.mediaStartMs < 0 || row.mediaEndMs <= row.mediaStartMs) return reject();
    const registration = cohort.get(row.personId);
    if (!registration || row.wallEndMs <= registration.verifiedAtMs || row.wallStartMs >= endMs) continue;
    const rows = segments.get(row.personId) ?? [];
    rows.push(row);
    segments.set(row.personId, rows);
  }
  const attendees = segments.size;
  result.attendanceRate = ratio(attendees, cohort.size);
  let wallMs = 0;
  const completed = new Set<string>();
  const durationValid = input.durationMs !== null && Number.isSafeInteger(input.durationMs) && input.durationMs > 0;
  for (const [person, rows] of segments) {
    const verifiedAt = cohort.get(person)!.verifiedAtMs;
    wallMs += unionLength(rows.map((row) => [Math.max(row.wallStartMs, verifiedAt, startMs), Math.min(row.wallEndMs, endMs)] as const));
    if (durationValid) {
      if (rows.some((row) => row.mediaEndMs > input.durationMs!)) return reject();
      const media = rows.map((row) => {
        const rate = (row.mediaEndMs - row.mediaStartMs) / (row.wallEndMs - row.wallStartMs);
        // Conservative inward millisecond rounding never invents coverage at a
        // reporting/verification boundary. Display rounding happens in the UI.
        return {
          startMs: Math.ceil(row.mediaStartMs + (Math.max(verifiedAt, startMs, row.wallStartMs) - row.wallStartMs) * rate),
          endMs: Math.floor(row.mediaStartMs + (Math.min(endMs, row.wallEndMs) - row.wallStartMs) * rate),
        };
      }).filter((row) => row.endMs > row.startMs);
      const completion = evaluateVideoCompletion(input.durationMs, media);
      if (completion.status === "available" && completion.completed) completed.add(person);
    }
  }
  result.averageViewingSeconds = ratio(wallMs / 1000, attendees, "seconds");
  result.completionRate = input.durationMs === null ? missing() : !durationValid ? invalid() : ratio(completed.size, attendees);
  if (!durationValid || input.completions === null || input.orders === null) return result;
  const firstCompletion = new Map<string, number>();
  for (const row of input.completions) {
    if (!id(row.personId) || !timestamp(row.firstCompletedAtMs)) return reject();
    const previous = firstCompletion.get(row.personId);
    if (previous !== undefined && previous !== row.firstCompletedAtMs) return reject();
    firstCompletion.set(row.personId, row.firstCompletedAtMs);
  }
  for (const row of input.orders) {
    if (!id(row.personId) || !timestamp(row.establishedAtMs) || typeof row.cancelled !== "boolean") return reject();
  }
  // A percentage alone cannot tell us the first threshold-crossing time.
  // Require the ingestion adapter's independently established completion fact.
  for (const person of completed) {
    const at = firstCompletion.get(person);
    if (at === undefined) return result;
    if (at < cohort.get(person)!.verifiedAtMs || at >= endMs) {
      result.postCompletionOrderRate = invalid();
      return result;
    }
  }
  const ordered = new Set<string>();
  for (const row of input.orders) {
    if (!completed.has(row.personId) || row.establishedAtMs >= endMs) continue;
    const eligibility = evaluatePostCompletionOrder(firstCompletion.get(row.personId)!, row);
    if (eligibility.status === "available" && eligibility.qualifies) ordered.add(row.personId);
  }
  result.postCompletionOrderRate = ratio(ordered.size, completed.size);
  // [S,T) excludes T, so a window ending exactly at the inclusive seventh day
  // has not yet observed its final boundary millisecond.
  result.provisionalConversions = [...completed].some((person) => endMs <= firstCompletion.get(person)! + POST_COMPLETION_ORDER_WINDOW_MS);
  return result;
}
/** Approved WP20 rules only. Callers must supply verified, same-person,
 * same-tenant, same-session/content-version facts. This module does not join
 * identities, infer missing events, or decide any metric denominator.
 */
export const COMPLETION_PERCENT = 90;
export const POST_COMPLETION_ORDER_WINDOW_MS = 7 * 24 * 60 * 60 * 1000;

type Unavailable = { status: "unavailable"; reason: "missing_data" | "invalid_data" };
export type MediaInterval = Readonly<{ startMs: number; endMs: number }>;

/** Intervals describe actually played media positions, not seek endpoints.
 * null means incomplete/unknown evidence; [] means a known empty history.
 * Replaying the same positions cannot increase unique coverage.
 */
export function evaluateVideoCompletion(
  durationMs: number | null,
  intervals: readonly MediaInterval[] | null,
): Unavailable | { status: "available"; coveredMs: number; completed: boolean } {
  if (durationMs == null || intervals == null) return { status: "unavailable", reason: "missing_data" };
  if (!Number.isSafeInteger(durationMs) || durationMs <= 0 || !Array.isArray(intervals) || Array.from(intervals).some((interval) =>
    !interval || typeof interval !== "object"
    || !Number.isSafeInteger(interval.startMs) || !Number.isSafeInteger(interval.endMs)
    || interval.startMs < 0 || interval.endMs < interval.startMs || interval.endMs > durationMs)) {
    return { status: "unavailable", reason: "invalid_data" };
  }
  const sorted = [...intervals].sort((a, b) => a.startMs - b.startMs);
  let coveredMs = 0;
  let end = 0;
  for (const interval of sorted) {
    coveredMs += Math.max(0, interval.endMs - Math.max(end, interval.startMs));
    end = Math.max(end, interval.endMs);
  }
  // Integer arithmetic avoids rounding a below-threshold viewer up to 90%.
  const completed = BigInt(coveredMs) * BigInt(100) >= BigInt(durationMs) * BigInt(COMPLETION_PERCENT);
  return { status: "available", coveredMs, completed };
}

export type EstablishedOrderFact = Readonly<{
  establishedAtMs: number | null;
  cancelled: boolean | null;
}>;

/** Completion time must come from verified first threshold-crossing evidence;
 * a total coverage number alone cannot establish when completion occurred.
 * The 7-day boundary is inclusive, using elapsed UTC milliseconds, not dates.
 * This is order eligibility, not paid conversion or mature cohort reporting.
 */
export function evaluatePostCompletionOrder(
  firstCompletedAtMs: number | null,
  order: EstablishedOrderFact,
): Unavailable | { status: "available"; qualifies: boolean } {
  if (firstCompletedAtMs == null || order == null || order.establishedAtMs == null || order.cancelled == null) {
    return { status: "unavailable", reason: "missing_data" };
  }
  if (typeof order.cancelled !== "boolean" || ![firstCompletedAtMs, order.establishedAtMs].every((value) => Number.isSafeInteger(value) && value >= 0 && value <= 8_640_000_000_000_000)) {
    return { status: "unavailable", reason: "invalid_data" };
  }
  const elapsed = order.establishedAtMs - firstCompletedAtMs;
  return { status: "available", qualifies: !order.cancelled && elapsed >= 0 && elapsed <= POST_COMPLETION_ORDER_WINDOW_MS };
}

      { scope, personId: "person-0", establishedAtMs: 94000, cancelled: false },
      { scope, personId: "person-1", establishedAtMs: 92000, cancelled: false },
    ],
  };
}
function value(metric: MetricValue): number {
  expect(metric.status).toBe("available");
  if (metric.status !== "available") throw new Error("Metric unavailable");
  return metric.value;
}

describe("approved five-metric reporting contract", () => {
  it("matches the proposal example without intermediate rounding or paid-order substitution", () => {
    const result = calculateCoreMetrics(fixture());
    expect(value(result.registrationRate)).toBe(40);
    expect(value(result.attendanceRate)).toBe(60);
    expect(value(result.completionRate)).toBeCloseTo(200 / 3, 10);
    expect(value(result.averageViewingSeconds)).toBeCloseTo(250 / 3, 10);
    expect(value(result.postCompletionOrderRate)).toBe(50);
    expect(result.unattributedRegistrations).toBe(1);
    expect(result.provisionalConversions).toBe(false);
  });
  it("deduplicates retransmissions, browsers and overlapping devices", () => {
    const input = fixture();
    const result = calculateCoreMetrics({ ...input, visits: [...input.visits!, ...input.visits!], registrations: [...input.registrations!, ...input.registrations!], viewing: [...input.viewing!, ...input.viewing!], completions: [...input.completions!, ...input.completions!], orders: [...input.orders!, ...input.orders!] });
    expect(result).toEqual(calculateCoreMetrics(input));
  });
  it("does not mutate caller arrays", () => {
    const input = fixture();
    const before = structuredClone(input);
    Object.freeze(input.viewing);
    Object.freeze(input.registrations);
    calculateCoreMetrics(input);
    expect(input).toEqual(before);
  });
  it("reports zero denominator distinctly from known zero conversion", () => {
    const empty = { ...fixture(), visits: [], registrations: [], viewing: [], completions: [], orders: [] };
    const result = calculateCoreMetrics(empty);
    for (const key of ["registrationRate", "attendanceRate", "completionRate", "averageViewingSeconds", "postCompletionOrderRate"] as const) expect(result[key]).toEqual({ status: "not_applicable", reason: "zero_denominator" });
    expect(value(calculateCoreMetrics({ ...fixture(), registrations: [], viewing: [], completions: [], orders: [] }).registrationRate)).toBe(0);
    expect(value(calculateCoreMetrics({ ...fixture(), viewing: [], completions: [], orders: [] }).attendanceRate)).toBe(0);
  });
  it.each(["visits", "registrations", "viewing", "completions", "orders"] as const)("never turns missing %s into zero", (key) => {
    const result = calculateCoreMetrics({ ...fixture(), [key]: null });
    const dependent = { visits: "registrationRate", registrations: "attendanceRate", viewing: "completionRate", completions: "postCompletionOrderRate", orders: "postCompletionOrderRate" } as const;
    expect(result[dependent[key]]).toEqual({ status: "unavailable", reason: "missing_data" });
  });
  it("keeps available attendance when a live recording has no known duration", () => {
    const result = calculateCoreMetrics({ ...fixture(), durationMs: null });
    expect(value(result.attendanceRate)).toBe(60);
    expect(result.completionRate.status).toBe("unavailable");
    expect(result.postCompletionOrderRate.status).toBe("unavailable");
  });
  it("does not attribute a registration to a later browser visit", () => {
    const input = fixture();
    const result = calculateCoreMetrics({ ...input, visits: input.visits!.map((row) => ({ ...row, visitedAtMs: 2500 })) });
    expect(value(result.registrationRate)).toBe(0);
    expect(result.unattributedRegistrations).toBe(5);
  });
  it("counts one browser once even when several people verify through it", () => {
    const input = fixture();
    expect(value(calculateCoreMetrics({ ...input, registrations: input.registrations!.map((row) => ({ ...row, browserId: "browser-0" })) }).registrationRate)).toBe(10);
  });
  it("excludes pre-window registrations and the exclusive right boundary", () => {
    const input = fixture();
    const result = calculateCoreMetrics({ ...input, startMs: 2000, endMs: 103000, registrations: [
      ...input.registrations!.slice(0, 3),
      { scope, personId: "older", browserId: null, verifiedAtMs: 1999 },
      { scope, personId: "later", browserId: null, verifiedAtMs: 103000 },
    ] });
    expect(value(result.attendanceRate)).toBe(100);
    expect(result.provisionalConversions).toBe(true);
    expect(result.unattributedRegistrations).toBe(3);
  });
  it("clips exposure at verification and report boundaries", () => {
    const input: CoreMetricsInput = { ...fixture(), endMs: 6000, durationMs: 10000,
      registrations: [{ scope, personId: "p", browserId: null, verifiedAtMs: 2000 }],
      viewing: [{ scope, personId: "p", wallStartMs: 1000, wallEndMs: 11000, mediaStartMs: 0, mediaEndMs: 10000 }], completions: [], orders: [] };
    const result = calculateCoreMetrics(input);
    expect(value(result.averageViewingSeconds)).toBe(4);
    expect(value(result.completionRate)).toBe(0);
  });
  it("counts replay time but not replayed media as extra completion coverage", () => {
    const input = fixture();
    const repeated = Array.from({ length: 5 }, (_, i) => ({ scope, personId: "p", wallStartMs: 10000 + i * 20000, wallEndMs: 30000 + i * 20000, mediaStartMs: 0, mediaEndMs: 20000 }));
    const result = calculateCoreMetrics({ ...input, registrations: [{ scope, personId: "p", browserId: null, verifiedAtMs: 2000 }], viewing: repeated, completions: [], orders: [] });
    expect(value(result.averageViewingSeconds)).toBe(100);
    expect(value(result.completionRate)).toBe(0);
  });
  it("preserves seek gaps and counts double speed by wall time", () => {
    const input = fixture();
    const result = calculateCoreMetrics({ ...input, registrations: [{ scope, personId: "p", browserId: null, verifiedAtMs: 2000 }], viewing: [
      { scope, personId: "p", wallStartMs: 3000, wallEndMs: 23000, mediaStartMs: 0, mediaEndMs: 40000 },
      { scope, personId: "p", wallStartMs: 33000, wallEndMs: 53000, mediaStartMs: 60000, mediaEndMs: 100000 },
    ], completions: [], orders: [] });
    expect(value(result.averageViewingSeconds)).toBe(40);
    expect(value(result.completionRate)).toBe(0);
  });
  it.each([[-1, 0], [0, 50], [POST_COMPLETION_ORDER_WINDOW_MS, 50], [POST_COMPLETION_ORDER_WINDOW_MS + 1, 0]])("order offset %i yields %i percent", (offset, expected) => {
    const result = calculateCoreMetrics({ ...fixture(), orders: [{ scope, personId: "person-0", establishedAtMs: 93000 + offset, cancelled: false }] });
    expect(value(result.postCompletionOrderRate)).toBe(expected);
  });
  it("excludes cancelled orders and orders at report cutoff", () => {
    const input = fixture();
    expect(value(calculateCoreMetrics({ ...input, orders: input.orders!.map((row) => ({ ...row, cancelled: true })) }).postCompletionOrderRate)).toBe(0);
    expect(value(calculateCoreMetrics({ ...input, endMs: 200000, orders: [{ scope, personId: "person-0", establishedAtMs: 200000, cancelled: false }] }).postCompletionOrderRate)).toBe(0);
  });
  it("marks the inclusive seventh-day boundary as immature until observed", () => {
    expect(calculateCoreMetrics({ ...fixture(), endMs: 93000 + POST_COMPLETION_ORDER_WINDOW_MS }).provisionalConversions).toBe(true);
    expect(calculateCoreMetrics({ ...fixture(), endMs: 93001 + POST_COMPLETION_ORDER_WINDOW_MS }).provisionalConversions).toBe(false);
  });
  it("requires first completion timing independently of coverage", () => {
    expect(calculateCoreMetrics({ ...fixture(), completions: [] }).postCompletionOrderRate).toEqual({ status: "unavailable", reason: "missing_data" });
  });
  it.each(["vendorId", "liveId", "contentVersionId", "mode"] as const)("rejects mixed %s instead of silently changing the denominator", (key) => {
    const input = fixture();
    const viewing = input.viewing!.map((row, i) => i ? row : { ...row, scope: { ...scope, [key]: key === "mode" ? "live" : "another" } as MetricScope });
    expect(calculateCoreMetrics({ ...input, viewing }).attendanceRate).toEqual({ status: "unavailable", reason: "invalid_data" });
  });
  it("rejects conflicting registration and completion facts", () => {
    const input = fixture();
    expect(calculateCoreMetrics({ ...input, registrations: [...input.registrations!, { ...input.registrations![0], verifiedAtMs: 5000 }] }).attendanceRate.status).toBe("unavailable");
    expect(calculateCoreMetrics({ ...input, completions: [...input.completions!, { ...input.completions![0], firstCompletedAtMs: 95000 }] }).postCompletionOrderRate.status).toBe("unavailable");
  });
  it.each([0, -1, NaN, Infinity, 1.5])("rejects invalid video duration %s", (durationMs) => {
    expect(calculateCoreMetrics({ ...fixture(), durationMs }).completionRate).toEqual({ status: "unavailable", reason: "invalid_data" });
  });
  it("rejects reversed intervals and coverage beyond the video", () => {
    const input = fixture();
    for (const patch of [{ wallEndMs: 1 }, { mediaEndMs: 100001 }, { mediaStartMs: -1 }]) {
      expect(calculateCoreMetrics({ ...input, viewing: [{ ...input.viewing![0], ...patch }] }).completionRate.status).toBe("unavailable");
    }
  });
});
